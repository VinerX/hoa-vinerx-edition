# Create Coalition of Khaz Modan
country_event = {
	id = ironforge.5
	title = ironforge.5.t
	desc = ironforge.5.d
	picture = GFX_magni_bronzebeard_event

	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = ironforge.5.option.1
		create_faction = "coalition_of_khaz_modan"

		set_country_flag = IRO_coalition_of_khaz_modan_cf

		# Send request to join Coalition
		GNO = {
			if = {
				limit = {
					NOT = { has_war_with = IRO }
				}
				country_event = ironforge.12
			}
		}
		NRT = {
			if = {
				limit = {
					NOT = { has_war_with = IRO }
				}
				country_event = ironforge.12
			}
		}

		hidden_effect = {
			news_event = { hours = 1 id = news.518 }
		}
	}
}

# Offer to join Coalition of Khaz Modan (sent from ironforge.5 to GNO and NRT)
country_event = {
	id = ironforge.12
	title = ironforge.12.t
	desc = ironforge.12.d
	picture = GFX_magni_bronzebeard_event

	is_triggered_only = yes

	option = { # Accept
		name = ironforge.12.option.1

		IRO = {
			add_to_faction = THIS.PREV
		}

		# Make peace with other members of the alliance (just in case GNO and NRT are at war for exemple)
		for_each_scope_loop = {
			array = allies
			white_peace = ROOT
		}

		ai_chance = {
			factor = 75
			modifier = {
				factor = 100
				OR = {
					tag = GNO
					tag = NRT
				}
				is_historical_focus_on = yes # Happens historically
			}
		}
	}

	option = { # Refuse
		name = ironforge.12.option.2

		ai_chance = {
			factor = 25
			modifier = {
				factor = 0
				OR = {
					tag = GNO
					tag = NRT
				}
				is_historical_focus_on = yes # Never happen historically
			}
		}
	}
}
