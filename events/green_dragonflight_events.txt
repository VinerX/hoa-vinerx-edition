country_event = { #add courption 1

	desc = greendragonflight.1.d

		}
	}

		name = greendragonflight.1.option.1
		GDF = {
			remove_ideas = GDF_emerald_nightmare_1
			add_ideas = GDF_emerald_nightmare_2
		}
	}
}

#turn into nightmare dragonflight
country_event = {
	id = greendragonflight.9
	title = greendragonflight.9.t
	desc = greendragonflight.9.d
	picture = GFX_nightmare_dragonflight_1_event
	
	fire_only_once = yes
	
	trigger = {
		tag = GDF
		old_gods > 0.8
	}

	option = {
		name = greendragonflight.9.option.1
		
		GDF = { leave_faction = yes }
		
		every_owned_state = {
			limit = { check_variable = { race = 50 } }
			set_variable = { race = 94 }
			add_core_of = NDF
			remove_core_of = GDF
		}
		
		NDF = {
			annex_country = { target = GDF transfer_troops = yes }
			change_tag_from = GDF
			set_politics = {
				ruling_party = old_gods
				elections_allowed = no
			}
			add_popularity = {
				ideology = old_gods
				popularity = 1
			}
			inherit_technology = GDF
		}
		
		RDF = { country_event = reddragonflight.1 }

		if = {
			limit = { 1355 = { is_controlled_by = GDS } }
			NDF = {
				annex_country = {
					target = GDS
					transfer_troops = yes
				}				
			}
		}
		if = {
			limit = { 1152 = { is_controlled_by = GDD }	}
			NDF = {
				annex_country = {
					target = GDD
					transfer_troops = yes
				}
			}
		}
		if = {
			limit = { 300 = { is_controlled_by = GTG } }
			NDF = {
				annex_country = {
					target = GTG
					transfer_troops = yes
				}
			}
		}
		if = {
			limit = { 126 = { is_controlled_by = GSR } }
			NDF = {
				annex_country = {
					target = GSR
					transfer_troops = yes
				}
			}
		}

		if = {
			limit = { 1355 = { owner = { has_government = old_gods } }	}
			1355 = { 
				transfer_state_to = NDF 
				add_core_of = NDF
			}
		}
		if = {
			limit = { 1152 = { owner = { has_government = old_gods } } }
			1152 = { 
				transfer_state_to = NDF 
				add_core_of = NDF
			}
		}
		if = {
			limit = { 300 = { owner = { has_government = old_gods } } }
			300 = { 
				transfer_state_to = NDF 
				add_core_of = NDF
			}
		}
		if = {
			limit = { 126 = { owner = { has_government = old_gods } } }
			126 = { 
				transfer_state_to = NDF 
				add_core_of = NDF
			}
			hidden_effect = {
				768 = { 
					transfer_state_to = NDF 
					add_core_of = NDF
				}
			}
		}
	}
}

#Give them their Old Religion Back
country_event = {
	id = greendragonflight.10
	title = greendragonflight.10.t
	desc = greendragonflight.10.d
	picture = GFX_nightmare_dragonflight_1_event

	immediate = {
		country_event = { id = greendragonflight.11 days = 1 }
	}
	
	fire_only_once = yes
	is_triggered_only = yes

	option = {
		name = greendragonflight.10.option.1
		if = {
			limit = { GDF = { has_idea = primary_religion_the_hour_of_twilight } }
			add_ideas = primary_religion_the_hour_of_twilight
		}
		if = {
			limit = { GDF = { has_idea = primary_religion_the_void } }
			add_ideas = primary_religion_the_void
		}
		if = {
			limit = { GDF = { has_idea = primary_religion_followers_of_ghuun } }
			add_ideas = primary_religion_followers_of_ghuun
		}
		if = {
			limit = { GDF = { has_idea = primary_religion_followers_of_yshaarj } }
			add_ideas = primary_religion_followers_of_yshaarj
		}	
		if = {
			limit = { GDF = { has_idea = primary_religion_followers_of_nzoth } }
			add_ideas = primary_religion_followers_of_nzoth
		}
		if = {
			limit = { GDF = { has_idea = primary_religion_followers_of_yogg_saron } }
			add_ideas = primary_religion_followers_of_yogg_saron
		}		
		if = {
			limit = { GDF = { has_idea = primary_religion_followers_of_cthun } }
			add_ideas = primary_religion_followers_of_cthun
		}
	}
}

#Teleport to Help deal with Red Dragons?
country_event = {
	id = greendragonflight.11
	title = greendragonflight.11.t
	desc = greendragonflight.11.d
	picture = GFX_nightmare_dragonflight_1_event

	immediate = {
		every_owned_state = {
			limit = { check_variable = { race = 50 } }
			#set_variable = { race = 94 }
			set_state_race_to_primary_race = yes
			set_state_religion_to_primary_religion = yes
			hidden_effect = {
				remove_core_of = GDF
				add_core_of = NDF	
			}
		}	
	}
	
	fire_only_once = yes
	is_triggered_only = yes

	option = {
		name = greendragonflight.11.option.1
		every_owned_state = {
			limit = { impassable = no }
			teleport_armies = {
				to_state = 202
				limit = { tag = NDF }
			}
		}
	}

	option = {
		name = greendragonflight.11.option.2

	}
}
