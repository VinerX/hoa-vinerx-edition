# ====================================================================
# PHASE 0: Alliance Joining — The Call from Capital City
# ====================================================================

# STR receives the call to formally join the Alliance, replacing the
# old grand_alliance.3 flow. The player chooses HOW Stromgarde commits.
country_event = {
	id = stromgarde_sw.1
	title = stromgarde_sw.1.t
	desc = stromgarde_sw.1.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event stromgarde_sw.1" }

	# Full commitment — send the bulk of the army to Lothar
	option = {
		name = stromgarde_sw.1.a
		join_grand_alliance = yes
		add_war_support = 0.05
		alliance_unity_raise_major = yes
		add_ideas = STR_army_divided_field
		set_country_flag = STR_sw_joined_field_army
		if = {
			limit = { LOR = { exists = yes } }
			LOR = { country_event = { id = lordaeron_second_war_proto.4 hours = 6 } }
		}
		ai_chance = {
			base = 60
			modifier = {
				factor = 2
				has_war = yes
			}
			modifier = {
				factor = 1.5
				is_historical_focus_on = yes
			}
		}
	}

	# Southern shield — keep forces to defend Arathi
	option = {
		name = stromgarde_sw.1.b
		join_grand_alliance = yes
		add_war_support = 0.03
		alliance_unity_raise = yes
		add_ideas = STR_army_divided_homeland
		set_country_flag = STR_sw_joined_homeland
		if = {
			limit = { LOR = { exists = yes } }
			LOR = { country_event = { id = lordaeron_second_war_proto.4 hours = 6 } }
		}
		ai_chance = {
			base = 25
			modifier = {
				factor = 1.5
				check_variable = { lordaeron_overextension > 2 }
			}
		}
	}

	# Mountain watch — prioritise intelligence on Alterac and passes
	option = {
		name = stromgarde_sw.1.c
		join_grand_alliance = yes
		add_war_support = 0.02
		alliance_unity_raise = yes
		add_ideas = STR_army_divided_mountain
		set_country_flag = STR_sw_joined_mountain
		set_country_flag = STR_alterac_investigation_started
		add_timed_idea = {
			idea = STR_suspicious_neighbour
			days = 365
		}
		if = {
			limit = { LOR = { exists = yes } }
			LOR = { country_event = { id = lordaeron_second_war_proto.4 hours = 6 } }
		}
		ai_chance = {
			base = 15
		}
	}

	# Refuse to join (historical off only)
	option = {
		name = stromgarde_sw.1.d
		add_political_power = 50
		alliance_unity_lower_medium = yes
		set_country_flag = STR_sw_refused_alliance
		if = {
			limit = { LOR = { exists = yes } }
			LOR = { country_event = { id = lordaeron_second_war_proto.4 hours = 6 } }
		}
		ai_chance = {
			base = 0
			modifier = {
				factor = 1
				NOT = { is_historical_focus_on = yes }
				has_government = neutral
			}
		}
	}
}

		name = stromgarde_sw.41.a

		alliance_unity_raise = yes
		add_ideas = STR_danath_expeditionary

	option = {
		name = stromgarde_sw.41.b
		set_country_flag = STR_danath_kept_home
		remove_ideas = STR_danath_rangers
		add_ideas = STR_danath_arathi_defender

			base = 25
		}
	}

	# Prepare southern expedition early
