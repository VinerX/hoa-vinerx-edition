# MONTHLY PULSE — QUE
################################################################################
country_event = {
	id = northern_war.2
	title = northern_war.2.t
	desc = northern_war.2.d
	picture = GFX_report_event_generic_factory

	is_triggered_only = yes
	hidden = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: northern_war.2 monthly pulse" }

	option = {
		name = northern_war.2.option

		# Cooldown clearing
		if = {
			limit = { has_country_flag = QUE_trap_cooldown }
			add_to_variable = { QUE_trap_cooldown_days = -1 }
			if = {
				limit = { check_variable = { QUE_trap_cooldown_days < 1 } }
				clr_country_flag = QUE_trap_cooldown
			}
		}
		if = {
			limit = { has_country_flag = QUE_channeling_sunwell }
			add_to_variable = { QUE_channel_cooldown = -1 }
			if = {
				limit = { check_variable = { QUE_channel_cooldown < 1 } }
				clr_country_flag = QUE_channeling_sunwell
			}
		}

		# Mana regeneration
		nw_mana_regen = yes

		# Barrier maintenance drain
		if = {
			limit = { has_country_flag = QUE_ban_dinoriel_active }
			nw_barrier_drain = yes
		}

		# Barrier cooldown tick
		if = {
			limit = {
				has_country_flag = QUE_ban_dinoriel_cooldown
				has_country_flag = QUE_ban_dinoriel_shattered
			}
			add_to_variable = { QUE_barrier_cooldown_remaining = -1 }
			if = {
				limit = {
					check_variable = { QUE_barrier_cooldown_remaining < 1 }
				}
				country_event = northern_war.23
			}
		}

		# Barrier falls if mana hits zero
		if = {
			limit = {
				has_country_flag = QUE_ban_dinoriel_active
				check_variable = { QUE_mana_reserve < 1 }
			}
			country_event = northern_war.22
		}

		# Mana display range flags (for decision panel indicator)
		clr_country_flag = QUE_mana_range_high
		clr_country_flag = QUE_mana_range_medium
		clr_country_flag = QUE_mana_range_low
		clr_country_flag = QUE_mana_range_critical
		if = {
			limit = { check_variable = { QUE_mana_reserve > 59 } }
			set_country_flag = QUE_mana_range_high
		}
		else_if = {
			limit = { check_variable = { QUE_mana_reserve > 29 } }
			set_country_flag = QUE_mana_range_medium
		}
		else_if = {
			limit = { check_variable = { QUE_mana_reserve > 9 } }
			set_country_flag = QUE_mana_range_low
		}
		else = {
			set_country_flag = QUE_mana_range_critical
		}

		# Zone modifier updates — remove old, apply new based on control
		# Southern zone
		if = {
			limit = { check_variable = { nw_southern_control > 1 } }
			if = { limit = { NOT = { has_idea = QUE_southern_defense } } add_timed_idea = { idea = QUE_southern_defense days = 45 } }
		}
		else = {
			if = { limit = { has_idea = QUE_southern_defense } remove_ideas = QUE_southern_defense }
		}

		# Runestone zone
		if = {
			limit = { check_variable = { nw_runestone_control > 1 } }
			if = { limit = { NOT = { has_idea = QUE_runestone_bastion } } add_timed_idea = { idea = QUE_runestone_bastion days = 45 } }
		}
		else = {
			if = { limit = { has_idea = QUE_runestone_bastion } remove_ideas = QUE_runestone_bastion }
		}

		# Farstrider zone
		if = {
			limit = { check_variable = { nw_farstrider_control > 2 } }
			if = { limit = { NOT = { has_idea = QUE_farstrider_patrols } } add_timed_idea = { idea = QUE_farstrider_patrols days = 45 } }
		}
		else = {
			if = { limit = { has_idea = QUE_farstrider_patrols } remove_ideas = QUE_farstrider_patrols }
		}

		# Approaches — evacuation logic
		if = {
			limit = { check_variable = { nw_approaches_enemy > 1 } }
			if = { limit = { NOT = { has_idea = QUE_approaches_threatened } } add_timed_idea = { idea = QUE_approaches_threatened days = 45 } }
		}
		else = {
			if = { limit = { has_idea = QUE_approaches_threatened } remove_ideas = QUE_approaches_threatened }
		}
	}
}
