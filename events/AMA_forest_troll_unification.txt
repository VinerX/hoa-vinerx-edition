	option = {
		name = ama_forest_unity.2.a
		trigger = {
			NOT = { is_puppet = yes }
		}
		leave_faction = yes
		AMA = {
			if = {
				limit = {
					is_faction_leader = yes
				}
				add_to_faction = ROOT
			}
			else = {
				add_to_faction = ROOT
			}
		}
		add_stability = 3
		add_popularity = {
			ideology = horde
			popularity = 0.10
		}
		set_country_flag = ama_tribe_joined_confederation
		AMA = {
			add_opinion_modifier = {
				target = ROOT
				modifier = large_increase
			}
		}
		ai_chance = {
			base = 70
		}
	}
	option = {
		name = ama_forest_unity.2.b
		add_stability = -2
		AMA = {
			create_wargoal = {
				type = puppet_wargoal_focus
				target = ROOT
			}
			add_opinion_modifier = {
				target = ROOT
				modifier = small_decrease
			}
		}
		ai_chance = {
			base = 30
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
		}
	}
}
