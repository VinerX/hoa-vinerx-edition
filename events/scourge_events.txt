			type = take_state_focus
			target = FHT
		}
  	}

# frenzyheart war cores
country_event = {
	id = scourge.2
	title = scourge.2.t
	desc = scourge.2.d

	fire_only_once = yes

	trigger = {
		tag = SRG
		SRG = {
			1748 = { is_fully_controlled_by = SRG }
			1746 = { is_fully_controlled_by = SRG }
		}
		NOT = { has_idea = SRG_dormant }
	}

	option = {
    	name = scourge.2.option.1
		load_oob = SRG_ghouls
		1748 = { 
			add_core_of = SRG 
			hidden_effect = {
				set_state_race_to_primary_race = yes
				set_state_religion_to_primary_religion = yes
			}
		}
		1746 = { 
			add_core_of = SRG
			hidden_effect = {
				set_state_race_to_primary_race = yes
				set_state_religion_to_primary_religion = yes
			} 
		}
		hidden_effect = {
			country_event = {
                id = scourge.3
                days = 30
            }
			1765 = {
				transfer_state_to = FHT
				add_core_of = FHT
			}
		}
		CAE = {
			country_event = {
				id = cult_of_the_damned.17
				days = 30
			}
		}
  	}
}
