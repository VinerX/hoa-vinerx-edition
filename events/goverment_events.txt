### City State

add_namespace = government_city_state

country_event = {
	id = government_city_state.1
	title = government_city_state.1.t
	desc = government_city_state.1.d

	hidden = yes

	trigger = {
		tag = ROOT
		ROOT = { 
			has_idea = city_state 
			has_war = no
		}
	}

	mean_time_to_happen = {
		days = 90
	}

	option = {
		name = government_city_state.1.option.1
		random_list = {
			20 = { country_event = government_city_state.2 }
			20 = { country_event = government_city_state.3 }
			20 = { country_event = government_city_state.4 }
			20 = { country_event = government_city_state.5 }
			20 = { country_event = government_city_state.6 }
		}
	}
}
