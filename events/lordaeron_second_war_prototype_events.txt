			days = 120
		}
		add_political_power = -40
		add_stability = -0.03
		LOR_second_war_proto_raise_alliance_unity = yes
		LOR_second_war_proto_raise_overextension = yes

		STR = {
			add_timed_idea = {
				idea = LOR_stromgarde_reinforcements_received
				days = 120
			}
			add_war_support = 0.03
			# Stromgarde is invited to formally join the Alliance
			# via the new interactive negotiation event.
			if = {
				limit = { NOT = { is_in_faction_with = STO } }
				country_event = { id = stromgarde_sw.1 hours = 6 }
			}
		}

		ai_chance = {
			base = 70
			modifier = {
				factor = 2
				STR = {
					has_war = yes
				}
			}
			modifier = {
				factor = 0.5
				check_variable = { lordaeron_overextension > 2 }
			}
			modifier = {
				factor = 0.25
				has_war_with = BRC
			}
		}
	}

	option = {
		name = lordaeron_second_war_proto.3.option.2

		add_timed_idea = {
			idea = LOR_preserved_royal_reserves
			days = 120
		}

		LOR_second_war_proto_lower_overextension = yes

			base = 30
			modifier = {
				factor = 2
				check_variable = { lordaeron_overextension > 2 }
			}
			modifier = {
				factor = 2
				has_war_with = BRC
			}
		}
	}
}
