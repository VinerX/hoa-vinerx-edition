	}

# Warsong (WSC) — Grom Hellscream
country_event = {
	id = draenor_reinforcements.10
	title = draenor_reinforcements.10.t
	desc = draenor_reinforcements.10.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = yes

	immediate = { log = "[GetDateText]: [WSC.GetName]: event draenor_reinforcements.10" }

	# Route threatened = additional narrative risk
	trigger = {
		exists = yes
		tag = WSC
	}

	option = {
		name = draenor_reinforcements.10.option.1
		ai_chance = {
			factor = 60
			modifier = {
				factor = 2
				BRC = { horde_authority_high = yes }
			}
			modifier = {
				factor = 0.3
				BRC = { horde_authority_defied = yes }
			}
		}
		# Warsong accepts — demands command role
		BRC = {
			if = {
				limit = { horde_authority_high = yes }
				# High authority: accepts immediately
				country_event = { id = draenor_reinforcements.30 hours = 6 }
				set_country_flag = clan_accepted_wsc
			}
			else_if = {
				limit = { horde_authority_medium = yes }
				# Medium: demands command role, costs authority
				country_event = { id = draenor_reinforcements.20 hours = 6 }
				set_country_flag = clan_negotiating_wsc
				set_country_flag = demand_offensive_command
			}
			else_if = {
				limit = { horde_authority_low = yes }
				# Low: may refuse (50% chance)
				random_list = {
					50 = {
						country_event = { id = draenor_reinforcements.20 hours = 6 }
						set_country_flag = clan_negotiating_wsc
						set_country_flag = demand_offensive_command
					}
					50 = {
						country_event = { id = draenor_reinforcements.91 hours = 6 }
					}
				}
			}
			else = {
				# Defied: likely refuses
				random_list = {
					30 = {
						country_event = { id = draenor_reinforcements.20 hours = 6 }
						set_country_flag = clan_negotiating_wsc
						set_country_flag = demand_offensive_command
					}
					70 = {
						country_event = { id = draenor_reinforcements.91 hours = 6 }
					}
				}
			}
		}
	}
}

	fire_only_once = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event draenor_reinforcements.20" }

	option = { # Accept demands
		name = draenor_reinforcements.20.option.1
		ai_chance = { factor = 65 }
		custom_effect_tooltip = draenor_reinforcements.20.option.1.tt
		if = {
			limit = { has_country_flag = demand_offensive_command }
			add_political_power = -50
			horde_authority_spend_small = yes
			set_country_flag = clan_accepted_wsc
		}
		else_if = {
			limit = { has_country_flag = demand_necrolyte_freedom }
