add_namespace = kul_council

# ============================================================
# Event .1 — Monthly pulse (hidden, syncs spirits, triggers events)
# ============================================================
country_event = {
	id = kul_council.1
	title = kul_council.1.t
	desc = kul_council.1.d
	picture = GFX_report_event_generic_generic
	hidden = yes

	is_triggered_only = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event kul_council.1 (monthly)"

		if = { limit = { NOT = { has_variable = kul_house_proudmoore } } set_variable = { kul_house_proudmoore = 0 } }
		if = { limit = { NOT = { has_variable = kul_house_waycrest } } set_variable = { kul_house_waycrest = 0 } }
		if = { limit = { NOT = { has_variable = kul_house_ashvane } } set_variable = { kul_house_ashvane = 0 } }
		if = { limit = { NOT = { has_variable = kul_house_stormsong } } set_variable = { kul_house_stormsong = 0 } }
		if = { limit = { NOT = { has_variable = kul_admiralty_stance } } set_variable = { kul_admiralty_stance = 0 } }

		# Clamp house influence
		if = { limit = { check_variable = { kul_house_proudmoore > 3 } } set_variable = { kul_house_proudmoore = 3 } }
		if = { limit = { check_variable = { kul_house_waycrest > 3 } } set_variable = { kul_house_waycrest = 3 } }
		if = { limit = { check_variable = { kul_house_ashvane > 3 } } set_variable = { kul_house_ashvane = 3 } }
		if = { limit = { check_variable = { kul_house_stormsong > 3 } } set_variable = { kul_house_stormsong = 3 } }

		# Sync spirits
		kul_council_sync_spirits = yes
		kul_council_sync_stance = yes

		# --- Contextual event triggers (random, once per trigger condition) ---

		# Dragonfire crisis: activate council dragon response event
		if = {
			limit = {
				is_in_faction_with = LOR
				has_war_with = BRC
				OR = {
					has_idea = KUL_sw_dragonfire_crisis_idea
					has_idea = KUL_sw_dragonfire_emergency_idea
				}
				NOT = { has_country_flag = KUL_council_dragon_event_fired }
			}
			set_country_flag = KUL_council_dragon_event_fired
			country_event = { id = kul_council.8 hours = 24 }
		}

		# Sea of Blood at risk — trigger crisis event
		if = {
			limit = {
				is_in_faction_with = LOR
				has_war_with = BRC
				NOT = { has_country_flag = KUL_council_sob_crisis_fired }
			}
			LOR = {
				if = {
					limit = { check_variable = { alliance_sea_control < 3 } }
					KUL = { set_country_flag = KUL_council_sob_crisis_fired country_event = { id = kul_council.10 hours = 24 } }
				}
			}
		}

		# SoB success — strengthen Daelin
		if = {
			limit = {
				is_in_faction_with = LOR
				has_war_with = BRC
				NOT = { has_country_flag = KUL_council_sob_success_fired }
			}
			LOR = {
				if = {
					limit = { check_variable = { alliance_sea_control > 4 } }
					KUL = { set_country_flag = KUL_council_sob_success_fired country_event = { id = kul_council.4 hours = 24 } }
				}
			}
		}

		# Random events (25% chance each, only fire once)
		random_list = {
			25 = {
				if = {
					limit = {
						is_in_faction_with = LOR
						has_war_with = BRC
						NOT = { has_country_flag = KUL_council_alliance_request_fired }
					}
					set_country_flag = KUL_council_alliance_request_fired
					country_event = { id = kul_council.5 hours = 24 }
				}
			}
			25 = {
				if = {
					limit = {
						is_in_faction_with = LOR
						has_war_with = BRC
						check_variable = { kul_house_stormsong > 0 }
						NOT = { has_country_flag = KUL_council_tidesage_warning_fired }
					}
					set_country_flag = KUL_council_tidesage_warning_fired
					country_event = { id = kul_council.6 hours = 24 }
				}
			}
			25 = {
				if = {
					limit = {
						is_in_faction_with = LOR
						has_war_with = BRC
						check_variable = { kul_house_ashvane > 0 }
						NOT = { has_country_flag = KUL_council_merchant_demand_fired }
					}
					set_country_flag = KUL_council_merchant_demand_fired
					country_event = { id = kul_council.11 hours = 24 }
				}
			}
			25 = {
			}
		}

		# Reset event flags periodically (every ~2 years of war)
		if = {
			limit = {
				is_in_faction_with = LOR
				has_war_with = BRC
				NOT = { has_country_flag = KUL_council_events_reset }
			}
			set_country_flag = KUL_council_events_reset
			clr_country_flag = KUL_council_dragon_event_fired
			clr_country_flag = KUL_council_sob_crisis_fired
			clr_country_flag = KUL_council_sob_success_fired
			clr_country_flag = KUL_council_alliance_request_fired
			clr_country_flag = KUL_council_tidesage_warning_fired
			clr_country_flag = KUL_council_merchant_demand_fired
		}

	}

	option = {
		name = kul_council.1.option.a
	}
}

		}
	}

		name = kul_council.3.option.c

		}

	}
