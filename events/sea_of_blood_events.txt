add_namespace = sea_of_blood

# Monthly sync — counts islands and updates tier spirits
country_event = {
	id = sea_of_blood.1
	title = sea_of_blood.1.t
	desc = sea_of_blood.1.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	hidden = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: Sea of Blood monthly sync" }

	option = {
		name = sea_of_blood.1.option.1
		update_sea_of_blood = yes
	}
}

# Tier changed — show notification to faction leader

	id = sea_of_blood.2
	title = sea_of_blood.2.t
	desc = sea_of_blood.2.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = no

	immediate = { log = "[GetDateText]: [Root.GetName]: Sea of Blood tier changed" }

	option = {
		name = sea_of_blood.2.option.1
		if = {
			limit = { NOT = { has_global_flag = news_sea_of_blood_begins } }

		}
	}
}
