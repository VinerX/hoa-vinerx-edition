# United Admiralty — the houses are bound
country_event = {
	id = kul_sw_exp.4
	title = kul_sw_exp.4.t
	desc = kul_sw_exp.4.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event kul_sw_exp.4" }

	option = {
		name = kul_sw_exp.4.option.a
		if = {
			limit = { has_idea = KUL_sw_house_waycrest_idea }
			remove_ideas = KUL_sw_house_waycrest_idea
		}
		if = {
			limit = { has_idea = KUL_sw_house_ashvane_idea }
			remove_ideas = KUL_sw_house_ashvane_idea
		}
		if = {
			limit = { has_idea = KUL_sw_house_stormsong_idea }
			remove_ideas = KUL_sw_house_stormsong_idea
		}
		add_ideas = KUL_sw_united_admiralty_idea
		add_political_power = 100
		add_stability = 0.10
		add_war_support = 0.05
		set_country_flag = KUL_united_admiralty_flag
	}
}
