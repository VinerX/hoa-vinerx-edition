# Victory or Defeat — the crossroads of the Second War
country_event = {
	id = amani_second_war.4
	title = amani_second_war.4.t
	desc = amani_second_war.4.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event amani_second_war.4" }

	option = {
		name = amani_second_war.4.option.1
		trigger = {
			has_war_with = QUE
			QUE = {
				OR = {
					has_capitulated = yes
				}
				surrender_progress > 0.5
			}
		}
		set_country_flag = ama_sw_horde_victory
		add_stability = 0.10
		add_war_support = 0.10
		add_political_power = 100
		ai_chance = {
			base = 10
		}
	}
	option = {
		name = amani_second_war.4.option.2
		trigger = {
			NOT = {
				has_war_with = QUE
				QUE = {
					OR = {
						has_capitulated = yes
					}
					surrender_progress > 0.5
				}
			}
		}
		set_country_flag = ama_sw_horde_defeated
		add_stability = 0.15
		add_war_support = -0.10
		add_popularity = {
			ideology = neutral
			popularity = 0.20
		}
		ai_chance = {
			base = 1
		}
	}
	option = {
		name = amani_second_war.4.option.3
		trigger = {
			has_war_with = QUE
		}
		set_country_flag = ama_sw_horde_victory
		add_stability = 0.05
		add_war_support = 0.05
		add_political_power = 50
		ai_chance = {
			base = 1
		}
	}
	option = {
		name = amani_second_war.4.option.4
		set_country_flag = ama_sw_horde_defeated
		add_stability = 0.20
		add_popularity = {
			ideology = neutral
			popularity = 0.15
		}
		ai_chance = {
			base = 1
		}
	}
}
