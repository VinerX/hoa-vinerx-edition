add_namespace = kultiras

# attending meeting of the seven kingdoms
country_event = {
	id = kultiras.1
	title = kultiras.1.t
	desc = kultiras.1.d

	is_triggered_only = yes

	fire_only_once = yes

	option = {
    name = kultiras.1.option.1
	add_popularity = { ideology = alliance popularity = 0.15 }
	add_opinion_modifier = {
		target = LOR
		modifier = attented_the_meeting_of_the_seven_kingdoms
	}
	reverse_add_opinion_modifier = {
		target = LOR
		modifier = attented_the_meeting_of_the_seven_kingdoms
	}
	add_timed_idea = {
		idea = LOR_meeting_of_the_seven_kingdoms_alliance
		days = 180
	}
	set_country_flag = attended_meeting_of_seven_kingdoms
		ai_chance = {
			base = 0
			modifier = {
				factor = 100
				is_historical_focus_on = yes
			}
			modifier = {
				factor = 50
				is_historical_focus_on = no
			}
			modifier = {
				factor = 50
				alliance > 49
			}
		}
  	}

	option = {
    name = kultiras.1.option.2
		ai_chance = {
			base = 0
			modifier = {
				factor = 50
				is_historical_focus_on = no
			}
			modifier = {
				factor = 50
				alliance < 50
			}
		}
  	}
}

# from secure_the_seas decision of LOR
country_event = {
	id = kultiras.2
	title = kultiras.2.t
	desc = kultiras.2.d

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		set_country_flag = accepted_alliance
		if = {
			limit = {
				LOR = { has_government = alliance }
			}
			set_politics = {
				ruling_party = alliance
				elections_allowed = no
			}
		}

		if = {
			limit = {
				LOR = { has_government = neutral }
			}
			set_politics = {
				ruling_party = neutral
				elections_allowed = no
			}
		}

		ai_chance = {
			base = 0
			modifier = {
				factor = 100
				is_historical_focus_on = yes
			}
			modifier = {
				factor = 50
				is_historical_focus_on = no
			}
			modifier = {
				factor = 50
				alliance > 49
			}
		}
	}

	option = {
		name = kultiras.2.option.1

		ai_chance = {
			base = 0
			modifier = {
				factor = 50
				is_historical_focus_on = no
			}
			modifier = {
				factor = 50
				alliance < 50
			}
			modifier = {
				factor = 200
				LOR = { has_government = neutral }
			}
		}
	}
}
