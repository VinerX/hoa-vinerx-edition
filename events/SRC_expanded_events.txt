add_namespace = SRC_expanded

# Blood Oath — Stormreaver commits to the Horde forever
country_event = {
	id = SRC_expanded.1
	title = SRC_expanded.1.t
	desc = SRC_expanded.1.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event SRC_expanded.1" }

	option = {
		name = SRC_expanded.1.option.1
		set_country_flag = SRC_blood_oath_sworn
		add_opinion_modifier = {
			target = BRC
			modifier = horde_loyalty
		}
		reverse_add_opinion_modifier = {
			target = BRC
			modifier = horde_loyalty
		}
		if = {
			limit = { BRC = { exists = yes } }
			BRC = {
				add_political_power = 50
				add_opinion_modifier = {
					target = SRC
					modifier = horde_loyalty
				}
			}
		}
		hidden_effect = {
			if = {
				limit = { has_country_flag = SRC_secret_agenda }
				clr_country_flag = SRC_secret_agenda
			}
			if = {
				limit = { has_idea = SRC_guldans_horde }
				remove_ideas = SRC_guldans_horde
			}
			if = {
				limit = { has_idea = SRC_guldans_horde_2 }
				remove_ideas = SRC_guldans_horde_2
			}
		}
	}
}

# Rise of Gorefiend — Teron takes command
country_event = {
	id = SRC_expanded.8
	title = SRC_expanded.8.t
	desc = SRC_expanded.8.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event SRC_expanded.8" }

	option = {
		name = SRC_expanded.8.option.1
		if = {
			limit = {
				has_character = SRC_terongor
			}
			add_country_leader_role = {
				character = SRC_terongor
				promote_leader = yes
				country_leader = {
					ideology = fel_type
				}
			}
		}
		add_ideas = SRC_gorefiends_council
		set_country_flag = SRC_gorefiend_risen
		if = {
			limit = { has_idea = SRC_raise_them_all }
			remove_ideas = SRC_raise_them_all
		}
		hidden_effect = {
			if = {
				limit = {
					has_country_flag = SRC_guldan_dead
					has_character = SRC_guldan
				}
				retire_character = SRC_guldan
			}
		}
	}
}
