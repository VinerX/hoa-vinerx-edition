country_event = { #unused event id
	id = dragonmaw.14
	title = dragonmaw.14.t
	desc = dragonmaw.14.d
	picture = GFX_report_event_german_parade_paris
	
	trigger = {
		tag = DMC
		always = no
		DMC = {	
			is_ai = no
			has_completed_focus = DMC_silvermoon_siege
			NOT = { has_idea = DMC_pending_punishment }	
			OR = {
				NOT = {	has_completed_focus = DMC_warchiefs_orders }
				NOT = {	has_completed_focus = DMC_double_efforts }
				NOT = {	has_completed_focus = DMC_a_dark_pact }
			}
		}
		country_exists = QUE
	}

	is_triggered_only = yes
	
	fire_only_once = yes
	
    option = {
        name = dragonmaw.14.option.1
		DMC = { leave_faction = yes }
		BRC = {
			declare_war_on = {
				target = DMC
				type = puppet_wargoal_focus
			}
		}
    }

	option = {
        name = dragonmaw.14.option.2
		kill_country_leader = yes
		add_stability = -0.10
		add_war_support = -0.05
		BRC = {
			if = {
				limit = { has_dlc = "Together for Victory" }
				set_autonomy = {
					target = DWC
					autonomous_state = autonomy_puppet
					freedom_level = 0.2
				}
				else = { puppet = DWC }
			}
		}
    }
}
