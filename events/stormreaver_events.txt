		add_to_array = { SRC_target_states = 350 }
		add_to_array = { SRC_target_states = 351 }
		add_to_array = { SRC_target_states = 352 }

		# Give the states owned by BRC
		for_each_scope_loop = {
			array = SRC_target_states
			if = {
				limit = { is_owned_by = BRC }
				add_core_of = SRC
				transfer_state_to = SRC
				remove_core_of = BRC

			BRC = {	add_to_faction = SRC }
