# Alliance Naval Council — Daelin offers the fleet
country_event = {
	id = kultiras_second_war.1
	title = kultiras_second_war.1.t
	desc = kultiras_second_war.1.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event kultiras_second_war.1" }

	option = {
		name = kultiras_second_war.1.option.1
		# Daelin commits the Kul Tiran fleet — Kul Tiras actually joins the Alliance.
		join_grand_alliance = yes
		LOR = { country_event = { id = grand_alliance.100 hours = 6 } }
		LOR = {
			add_opinion_modifier = {
				target = KUL
				modifier = STR_friendly_relations
			}
		}
		reverse_add_opinion_modifier = {
			target = LOR
			modifier = STR_friendly_relations
		}
		LOR = {
			add_opinion_modifier = {
				target = KUL
				modifier = medium_increase
			}
		}
		reverse_add_opinion_modifier = {
			target = LOR
			modifier = medium_increase
		}
		every_country = {
			limit = {
				is_in_faction_with = LOR
				NOT = { tag = KUL }
				NOT = { tag = LOR }
			}
			add_opinion_modifier = {
				target = KUL
				modifier = small_increase
			}
			reverse_add_opinion_modifier = {
				target = PREV
				modifier = small_increase
			}
		}
		add_war_support = 0.10
		add_political_power = 75
		LOR = {
			add_war_support = 0.05
		}
	}
}
