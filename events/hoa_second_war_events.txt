				transfer_state = 103

			if = {
				limit = {
					FWC = { is_in_faction_with = BRC }
				}
				FWC = {
					annex_country = {
					    target = ALT
					    transfer_troops = no
					}
				}
			}
			# Else just transfer to BRC
			else = {
				BRC = {

					}
				}
			}
		}

		# If Tol'barad (TOL) is allied with LOR (via STR)
		# AND Stormreavers (SRC) are part of the Horde
		if = {
			limit = {
				TOL = { is_in_faction_with = LOR }
				SRC = { is_in_faction_with = BRC }
			}
			# Stormreavers (SRC) annex Tol'Barad
			SRC = {
				annex_country = {
				    target = TOL

				}
			}

					    transfer_troops = no
					}
				}

		# If Stromgarde (STR) is allied with LOR

			if = {

		}

		# If Dalaran (DAL) is allied with LOR
		if = {
			limit = {
				DAL = { is_in_faction_with = LOR }
			}
			# BRC annex DAL
			annex_country = {
			    target = DAL
			    transfer_troops = no
			}
			# If Alterac is part of the Horde
			if = {
				limit = {
					ALT = { is_in_faction_with = BRC }
				}
				# Transfer Dalaran outskirt to ALT
				ALT = { transfer_state = 2218 }
			}
		}

		# If Stormwind Refugees (STO) is allied with LOR
		if = {
			limit = {
				STO = { is_in_faction_with = LOR }
			}
			# BRC annex STO
			annex_country = {
			    target = STO
			    transfer_troops = no
			}
			# If Stormreavers (SRC) are part of the Horde
			if = {
				limit = {
					SRC = { is_in_faction_with = BRC }
				}
				# Transfer Newman's Landing
				SRC = { transfer_state = 1576 }
			}
		}

		# If Gilneas (GIL) is allied with LOR
		if = {
			limit = {
				GIL = { is_in_faction_with = LOR }
			}
			# BRC annex GIL
			annex_country = {
			    target = GIL
			    transfer_troops = no
			}
			# If Stormreavers (SRC) are part of the Horde

				}
				# SRC annex KUL
				SRC = {

					    target = KUL
					    transfer_troops = no
					}
				}
			}
			else = {
				# BRC annex KUL
				SRC = {
					annex_country = {

					}
				}
			}
		}

		# BRC annex all of Lordaeron

			}
		}
		# Else if Dragonmaw (DMC) is part of the Horde
		else_if = {
			limit = {
				DMC = { is_in_faction_with = BRC }
			}
			# Give DMC the east coast of Lordaeron (Tyr's Hand, Hasic, etc...)
			DMC = {
				transfer_state = 30
				transfer_state = 226
				transfer_state = 227
				transfer_state = 228
				transfer_state = 229

				transfer_state = 235

# Alliance Victory — Blackrock defeated in Second War (scripted peace)
country_event = {
	id = second_war.2
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = { log = "[GetDateText]: Alliance Victory in Second War" }

	option = {
		news_event = { id = second_war_news.37 hours = 6 }
		news_event = { id = second_war_news.36 hours = 24 }
		news_event = { id = second_war_news.38 hours = 72 }
		# Stromgarde: Veterans of the Second War
		if = {
			limit = {
				STR = {
					exists = yes
					is_in_faction_with = LOR
				}
			}
			STR = { country_event = { id = stromgarde_sw.50 hours = 6 } }
		}
	}
}  

# NEWS EVENTS (50+)
# End of Second War (Alliance of Lordaeron Defeated)
news_event = {
	id = second_war.50
	title = second_war.50.t
	desc = second_war.50.d
	picture = GFX_news_event_fall_of_lordaeron_city

	is_triggered_only = yes
	fire_only_once = yes
	major = yes # Trigger for everyone

	option = { # Horde option
		name = second_war.50.option.1

		trigger = {
			has_government = horde
		}
	}

	option = { # Alliance option (for anyone that did not join the war and / or is left alive)
		name = second_war.50.option.2

		trigger = {
			has_government = alliance
		}
	}
}
