# Rise of the Horde (year 581) religion setup
country_event = {
	id = state_religion_setup.581
	title = state_religion_setup.581.t
	desc = state_religion_setup.581.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = yes
	hidden = yes

 	option = {
 		name = state_religion_setup.581.a

		every_country = {
            limit = {
                NOT = { # Exclude non playable countries
                    tag = ZZD
                    tag = ZZO
                    tag = ZZT
                    tag = ZZW
                }
            }
            every_owned_state = {
                limit = {
                    NOT = { # Excludes states that are not of the owner's primary religion (see below)
						state = 112
						state = 135
						state = 203
						state = 316
						state = 356
						state = 367
						state = 432
						state = 473
						state = 1348
						state = 1360
						state = 1368
						state = 1385
						state = 1394
						state = 1445
						state = 1453
						state = 1489
						state = 1490
						state = 1491
						state = 1493
						state = 1495
						state = 1496
						state = 1512
						state = 1578
						state = 1597
						state = 2218
                    }
                }
                meta_effect = { # Set state race to the owner's primary religion
        			text = {
        				set_variable = { religion = [MAIN_RELIGION] }
        			}
        			MAIN_RELIGION = "[PREV.GetMainReligionVariable]"
        		}
            }

			# Setup the state religion for the states that have been excluded (see above limit block)
	        # If modifying a state religion, don't forget it to add it above in the exclude block! (it will still work otherwise, but it's better for optimization)
	        # Order by religion ID and then state ID

	        # CHURCH OF THE HOLY LIGHT 1
	        2218 = { set_variable = { var = religion value = 1 } }

			# ARCANE STUDIES 12
			203 = { set_variable = { var = religion value = 12 } }
			316 = { set_variable = { var = religion value = 12 } }

			# THE SUNWELL 13
			135 = { set_variable = { var = religion value = 13 } }
			473 = { set_variable = { var = religion value = 13 } }

			# DRUIDISM 15
			1360 = { set_variable = { var = religion value = 15 } }
