add_namespace = warsong_third_war

# siege of durnholde
country_event = {
    id = warsong_third_war.1
    title = warsong_third_war.1.t
    desc = warsong_third_war.1.d
	picture = GFX_durnholde_ruins_event

	is_triggered_only = yes

	fire_only_once = yes

    option = { # Siege of Durnholde Part 1
        name = warsong_third_war.1.option.1

		WSC = {
			transfer_state = 984
			transfer_state = 979
			transfer_state = 985
			transfer_state = 986
			transfer_state = 596
			transfer_state = 600
			transfer_state = 608
			set_capital = { state = 984 }
			every_state = {
				limit = {
					impassable = no
					owner = { NOT = { has_war_with = WSC } }
				}
				teleport_armies = {
					to_state = 984
					limit = { tag = WSC }
				}
			}
			add_ideas = WSC_isolated
		}

		hidden_effect = {
			124 = { set_demilitarized_zone = yes }
			1542 = { set_demilitarized_zone = yes }
			2215 = { set_demilitarized_zone = yes }
		}
		2215 = { add_manpower = -11000 }
		985 = {	add_manpower = 11000 }
		random_army_leader = {
			gain_xp = 100
			add_attack = 1
		}
		hidden_effect = {
			WSC = {	country_event = { id = warsong_third_war.2 days = 7 } }
		}
    }
}
