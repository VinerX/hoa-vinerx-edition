# From shadow_council.2.option.1
country_event = {
	id = darkiron.1
	title = darkiron.1.t
	desc = darkiron.1.d
	picture = GFX_cho_gall_event

	is_triggered_only = yes

	fire_only_once = yes

    option = { # Accept offer (concede territories)
        name = darkiron.1.option.1

		if = {
			limit = { DIC = { NOT = { has_completed_focus = DIC_The_Masters_Will } } }
			complete_national_focus = DIC_The_Masters_Will
		}

		add_political_power = 100

		add_to_variable = {
			var = faith_points
			value = 250
			tooltip = add_faith_points_tt
		}

		add_popularity = {
		    ideology = neutral
		    popularity = 0.05
		}

		if = {
			limit = { NOT = { has_non_aggression_pact_with = TSC } }
			diplomatic_relation = {
				country = TSC
				relation = non_aggression_pact
				active = yes
			}	
		}

		TSC = {	country_event = { id = shadow_council.3 } }
		custom_effect_tooltip = darkiron.1.option.1.tt

		ai_chance = {
			base = 90

			modifier = {
				factor = 110
				is_historical_focus_on = yes
			}
		}
	}

    option = { # Refuse
        name = darkiron.1.option.2

		add_stability = -0.05

		add_political_power = -25

		add_popularity = {
		    ideology = old_gods
		    popularity = -0.25
		}

		ai_chance = {
			base = 10
			modifier = {
				factor = -110
				is_historical_focus_on = yes
			}
		}
	}

}

# Blackrock Offer (from BRC_alliance_with_the_dark_iron focus and blackrock.22 event)
country_event = {
	id = darkiron.2
	title = darkiron.2.t # Default title
	picture = GFX_cho_gall_event

	is_triggered_only = yes

	fire_only_once = yes

	# BRC ask for non-aggression pact
	title = {
		text = darkiron.2.t.1
		trigger = {
			BRC = { has_country_flag = BRC_ask_DIC_for_non_aggression_flag }
		}
	}

	desc = {
		text = darkiron.2.d.1
		trigger = {
			BRC = { has_country_flag = BRC_ask_DIC_for_non_aggression_flag }
		}
	}

	# BRC ask for guarentee
	title = {
		text = darkiron.2.t.2
		trigger = {
			BRC = { has_country_flag = BRC_ask_DIC_for_guarantee_flag }
		}
	}

	desc = {
		text = darkiron.2.d.2
		trigger = {
			BRC = { has_country_flag = BRC_ask_DIC_for_guarantee_flag }
		}
	}

    option = { # Accept
        name = darkiron.2.option.1

		if = {
			limit = {
				BRC = { has_country_flag = BRC_ask_DIC_for_non_aggression_flag }
			}
			diplomatic_relation = {
			    country = BRC
			    relation = non_aggression_pact
			    active = yes
			}
		}
		else_if = {
			limit = {
				BRC = { has_country_flag = BRC_ask_DIC_for_guarantee_flag }
			}
			diplomatic_relation = {
			    country = BRC
			    relation = guarantee
			    active = yes
			}
		}

		BRC = {
			clr_country_flag = BRC_ask_DIC_for_non_aggression_flag
			clr_country_flag = BRC_ask_DIC_for_guarantee_flag

			country_event = {
				id = generic_event.1 # Generic accept event
				days = 1
			}
		}

		hidden_effect = {
			# Get whispers about giving BRC the Badlands
			# This is to prevent border gore and allow the Burning Blade Clan to move there
			country_event = {
				id = darkiron.3
				days = 7
			}
		}

		ai_chance = {
			base = 90

			modifier = {
				factor = 110
				is_historical_focus_on = yes
			}
		}
	}

    option = { # Refuse
        name = darkiron.2.option.2

		BRC = {
			clr_country_flag = BRC_ask_DIC_for_non_aggression_flag
			clr_country_flag = BRC_ask_DIC_for_guarantee_flag

			country_event = {
				id = generic_event.2 # Generic refusal event
				days = 1
			}
		}

		ai_chance = {
			base = 10
			modifier = {
				factor = -110
				is_historical_focus_on = yes
			}
		}
	}
}
