# Blackrock deal with Alterac
country_event = {
	id = alterac.4
	title = alterac.4.t
	desc = alterac.4.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes

	fire_only_once = yes

    option = {
        name = alterac.4.option.1

        BRC = {
            country_event = blackrock.40
        }

        every_enemy_country = {
            limit = {
                is_in_faction_with = BRC
            }
            white_peace = ALT
        }

        diplomatic_relation = {
            country = BRC
            relation = non_aggression_pact
            active = yes
        }

        ai_chance = {
			factor = 50
			modifier = {
				factor = 150
				is_historical_focus_on = yes # Happens historically
			}
		}
    }

    option = {
        name = alterac.4.option.2

        BRC = { country_event = blackrock.41 }

        ai_chance = {
			factor = 50
			modifier = {
				factor = -150
				is_historical_focus_on = yes # Does not happen historically
			}
		}
    }
}
