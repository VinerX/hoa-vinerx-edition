# Deal with the trade Prince (from BRC_deal_with_the_trade_prince)
country_event = {
	id = steamwheedle.1
	title = steamwheedle.1.t
	desc = steamwheedle.1.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes

	fire_only_once = yes

    option = {
        name = steamwheedle.1.option.1

		# Accept offer (aka. join the Horde)
		BRC = {
			add_to_faction = STC
			country_event = generic_event.1 # Generic accept event
		}

		STC = {
			add_ideas = BRC_stormwinds_treasury_idea
			set_politics = {
				ruling_party = horde
				elections_allowed = no
			}
			add_popularity = {
				ideology = horde
				popularity = 1
			}
		}

		BOO = {
			add_ideas = BRC_stormwinds_treasury_idea
			set_politics = {
				ruling_party = horde
				elections_allowed = no
			}
			add_popularity = {
				ideology = horde
				popularity = 1
			}
		}

		if = {
			limit = { STC = { controls_state = 818 } }
			818 = {
				add_extra_state_shared_building_slots = 3
				add_building_construction = {
					type = dockyard
					level = 3
					instant_build = yes
				}
			}
		}

		if = {
			limit = { BOO = { controls_state = 46 } }
			46 = {
				add_extra_state_shared_building_slots = 3
				add_building_construction = {
					type = dockyard
					level = 3
					instant_build = yes
				}
			}
		}

		ai_chance = {
			base = 1
		}
	}

	# Refuse (aka. not join the Horde)
    option = {
        name = steamwheedle.1.option.2

		BRC = {
			country_event = generic_event.2 # Generic refusal event
		}
		ai_chance = {
			base = 0
			modifier = {
				factor = 1
				is_historical_focus_on = no
			}
		}
	}
}
