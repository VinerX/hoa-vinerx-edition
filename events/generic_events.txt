# Generic offer for mutual guarantee
country_event = {
  id = generic_event.4
  title = generic_event.4.t
  desc = generic_event.4.d

  is_triggered_only = yes

  # Accept
  option = {
      name = generic_event.4.option.1

      # Give THIS and FROM a mutual guarantee
      THIS = {
          diplomatic_relation = {
            country = FROM
            relation = guarantee
            active = yes
          }
      }
      FROM = {
          diplomatic_relation = {
            country = THIS
            relation = guarantee
            active = yes
          }
      }

      # Tell FROM that THIS have accepted
      FROM = { country_event = generic_event.1 } # Generic accept event
  }

  # Refuse
  option = {
      name = generic_event.4.option.2

      # Tell FROM that THIS have refused
      FROM = { country_event = generic_event.2 } # Generic refusal event
  }
}

country_event = {
	id = generic_event.6
	title = generic_event.6.t
	desc = generic_event.6.d
	
	is_triggered_only = yes

	option = {
		name = generic_event.6.option.1

    if = {
			limit = {
				has_core_occupation_modifier = {
					occupied_country_tag = FROM.FROM
					modifier = compliance_80
				}
				NOT = {
					any_country_with_original_tag = {
						original_tag_to_check = FROM.FROM
						is_puppet_of = PREV
						has_autonomy_state = autonomy_collaboration_government
					}
				}
			}
			set_temp_variable = { country_to_initiate = FROM.FROM }
			instantiate_collaboration_government = yes
		}
		
		ai_chance = {
			base = 0
		}
	}

	option = {
		name = generic_event.6.option.2
		ai_chance = {
			base = 100
		}
	}
}
