add_namespace = horde_authority_events

# ============================================================
# Warchief Authority events — Horde faction-wide mechanics
# Triggered by focus trees, decisions, and other events
# to reflect the Warchief's grip on the clans.
# ============================================================

# 1: A clan submits to the Warchief (+10 authority)
# Triggered from focus trees when a clan pledges loyalty
country_event = {
	id = horde_authority_events.1
	title = horde_authority_events.1.t
	desc  = horde_authority_events.1.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event horde_authority_events.1 (clan submits)" }

	option = {
		name = horde_authority_events.1.a
		horde_authority_raise_medium = yes
	}
}

# 6: The Warchief Commands — spend authority for a massive offensive
# Triggered from decision when authority is high enough
country_event = {
	id = horde_authority_events.6
	title = horde_authority_events.6.t
	desc  = horde_authority_events.6.d
	picture = GFX_report_event_generic_fascist_split

	is_triggered_only = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event horde_authority_events.6 (total commitment)" }

	option = {
		name = horde_authority_events.6.a
		trigger = {
			horde_authority_can_spend_large = yes
		}
		ai_chance = { factor = 85 }
		horde_authority_spend_large = yes
		every_country = {
			limit = { is_in_faction_with = BRC }
			add_timed_idea = {
				idea = horde_authority_total_commitment
				days = 90
			}
		}
	}
}

# 8: Doomhammer crushes dissent — demonstrate power to restore authority
# Available when authority is low and BRC has PP to spend
country_event = {
	id = horde_authority_events.8
	title = horde_authority_events.8.t
	desc  = horde_authority_events.8.d
	picture = GFX_report_event_coup

	is_triggered_only = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event horde_authority_events.8 (crush dissent)" }

	# Option A: Make an example (+10 authority, costs PP, loses stability)
	option = {
		name = horde_authority_events.8.a
		ai_chance = { factor = 70 }
		add_political_power = -30
		add_stability = -0.03
		horde_authority_raise_medium = yes
	}

	# Option B: Let it go for now (-5 authority, preserve stability)
	option = {
		name = horde_authority_events.8.b
		ai_chance = { factor = 30 }
		horde_authority_lower = yes
		add_stability = 0.02
	}
}
