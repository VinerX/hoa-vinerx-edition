add_namespace = grand_alliance

# ============================================================
# Grand Alliance formation — invitation events (Second War)
#
# Lordaeron formally invites the remaining historical kingdoms into the
# Alliance of Lordaeron. Accepting actually adds the country to the faction
# (see join_grand_alliance / merge_coalition_of_khaz_modan scripted effects).
# AI chances are weighted heavily toward the historical outcome (accept).
# ============================================================

# ----- Quel'Thalas -----
country_event = {
	id = grand_alliance.1
	title = grand_alliance.1.t
	desc = grand_alliance.1.d
	picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event grand_alliance.1 (QUE invite)" }

	option = { # Accept — join the Alliance
		name = grand_alliance.1.a
		ai_chance = {
			base = 90
			modifier = {
				factor = 2
				has_opinion = { target = LOR value > 40 }
			}
		}
		join_grand_alliance = yes
		add_war_support = 0.05
		LOR = { country_event = { id = grand_alliance.100 hours = 6 } }
	}
	option = { # Decline
		name = grand_alliance.1.b
		ai_chance = { base = 10 }
		add_political_power = 25
	}
}
