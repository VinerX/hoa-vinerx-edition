# Alliance relief arrives / final war machine — triggered by alliance war machine focus
country_event = {
	id = gnomeregan_second_war.3
	title = gnomeregan_second_war.3.t
	desc = gnomeregan_second_war.3.d
	picture = GFX_report_event_alliance_victory

	is_triggered_only = yes
	fire_only_once = yes

	immediate = { log = "[GetDateText]: [Root.GetName]: event gnomeregan_second_war.3" }

	option = {
		name = gnomeregan_second_war.3.option.1
		add_war_support = 0.10
		add_stability = 0.05
		set_temp_variable = { var_name = gno_war_participants value = 0 }
		if = {
			limit = { IRO = { has_war_with = BRC } }
			IRO = { add_opinion_modifier = { target = GNO modifier = GNO_sw_debt_of_gratitude } }
			set_temp_variable = { var_name = gno_war_participants value = 1 }
		}
		if = {
			limit = { LOR = { has_war_with = BRC } }
			LOR = { add_opinion_modifier = { target = GNO modifier = GNO_sw_debt_of_gratitude } }
		}
	}
}
