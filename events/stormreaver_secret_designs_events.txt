	id = src_secret_designs.1
	title = src_secret_designs.1.t
	desc = src_secret_designs.1.d
	picture = GFX_report_event_orc_chiefs

	fire_only_once = no

	trigger = {
		tag = SRC
		is_in_faction_with = BRC
		check_variable = { guldan_suspicion > 39 }
		NOT = { has_country_flag = SRC_betrayed_horde }
		NOT = { has_country_flag = SRC_guldan_dead }
	}

	mean_time_to_happen = {
		days = 240
		modifier = {
			factor = 0.5
			check_variable = { guldan_suspicion > 59 }
		}
		modifier = {
			factor = 0.5
			src_authority_high_constraint = yes
		}
		modifier = {
			factor = 2.0
			check_variable = { guldan_suspicion < 50 }
		}
	}

	immediate = { log = "[GetDateText]: [Root.GetName]: event src_secret_designs.1" }

	option = {
		name = src_secret_designs.1.a
		horde_authority_lower = yes
		custom_effect_tooltip = src_secret_designs.1.a.tt
		src_suspicion_add_medium = yes
	}
	option = {
		name = src_secret_designs.1.b
		src_suspicion_reduce_small = yes
		horde_authority_raise = yes
		BRC = {
			add_opinion_modifier = {
				target = SRC
				modifier = small_increase
			}
		}
	}
	option = {
		name = src_secret_designs.1.c
		src_suspicion_add_large = yes
		horde_authority_lower_medium = yes
