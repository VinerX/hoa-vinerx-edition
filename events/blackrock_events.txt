					limit = {
						state = 368

						}

						type = industrial_complex
						level = 1
					}

							type = industrial_complex
							level = 1
						}

					chance = 25

# Ban Fel Magic (Blackrock side event)
country_event = {
	id = blackrock.15
	title = blackrock.15.t
	desc = blackrock.15.d
	picture = GFX_orgrim_doomhammer_event
	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = blackrock.15.option.1

		remove_ideas = primary_religion_demonology
		remove_ideas = secondary_religion_necrolytes
		add_ideas = primary_religion_necrolytes
		add_ideas = BRC_banned_fel_magic

		# Remove Demonology (id 27) from accepted_religions array
		# Should make scripted effects for that eventually
		remove_from_array = {
			array = accepted_religions
			value = 27
		}

		custom_effect_tooltip = BRC_ban_fel_magic_states_tt
		hidden_effect = {
			capital_scope = {
				set_state_race_to_primary_race = yes
				set_state_religion_to_primary_religion = yes
			}

			# Loop through owned states
			for_each_scope_loop = {
				array = owned_states
				# 75% chance to be converted
				if = {
					limit = {
						# If state is demonology
						check_variable = { religion = 27 }
					}
					random = {
						chance = 75
						set_state_religion_to_primary_religion = yes
					}
				}
			}
		}

		custom_effect_tooltip = BRC_ban_fel_magic_allies_tt
		hidden_effect = {
			every_country = {
				if = {
					limit = {
						is_in_faction_with = BRC
