# Serpents in the North (True Devotion Event)
country_event = {
	id = zemlan.7
	title = zemlan.7.t
	desc = zemlan.7.d
	#picture =

	is_triggered_only = yes

	fire_only_once = yes

	# No exceptions for the Sethrak. Land on the Forgotten Marshes and prepare to fight them later
	option = {
		name = zemlan.7.option.1
		add_timed_idea = {
			idea = ZLC_raiding_parties
			days = 30
		}
		2029 = {
			OWNER = {
				ROOT = {
					create_wargoal = {
						target = PREV
						type = take_state_focus
						generator = { 2029 }
					}
				}
			}
		}
		add_opinion_modifier = {
			target = DEV
			modifier = ZLC_sethrak_hatred
		}
		DEV = {
			add_opinion_modifier = {
				target = ZLC
				modifier = ZLC_raided_the_marsh
			}
		}
		custom_effect_tooltip = ZLC_mutiny_consequences_tt
		set_country_flag = ZLC_mutiny_antagonize_devoted
	}

	# Offer a pact to The Devoted against The Faithless Empire
	option = {
		name = zemlan.7.option.2
		if = {
			limit = {
				country_exists = DEV
			}
			DEV = {
				#Offer The Devoted a non-aggression pact
				country_event = zemlan.8
			}
		}
		else_if = {
			limit = {
				any_owned_state = {
					is_core_of = DEV
				}
			}
			every_owned_state = {
				limit = {
					is_core_of = DEV
				}
				add_compliance = 20
				add_resistance_target = -20
				add_state_modifier = {
					modifier = {
						local_non_core_manpower = 0.5
					}
				}
			}
		}
		else = {
			add_stability = 0.05
			add_manpower = 250
		}
		custom_effect_tooltip = ZLC_mutiny_consequences_tt
		set_country_flag = ZLC_mutiny_ally_devoted
	}
}
