# ===========================================================
# Khaz Modan Siege — Events
# ===========================================================
add_namespace = khaz_modan_siege

# --- Weekly synchronisation event (hidden) ---
country_event = {
	id = khaz_modan_siege.1
	hidden = yes
	is_triggered_only = yes

	trigger = {
		has_war = yes
		OR = {
			tag = IRO
			tag = GNO
			tag = BHC
		}
	}

	immediate = {
		if = {
			limit = { tag = IRO }
			khaz_modan_ironforge_weekly = yes
		}
		if = {
			limit = { tag = GNO }
			khaz_modan_gnomeregan_weekly = yes
		}
		if = {
			limit = { tag = BHC }
			khaz_modan_bhc_weekly = yes
		}
	}
}

# .10 — The Mountain Stands Alone (siege begins)
country_event = {
	id = khaz_modan_siege.10
	title = khaz_modan_siege.10.t
	desc = khaz_modan_siege.10.d
	picture = GFX_report_event_german_parade_paris
	fire_only_once = yes

	trigger = {
		tag = IRO
		has_war = yes
	}

	immediate = {
		log = "[GetDateText]: [Root.GetName]: Ironforge siege has begun"
		khaz_modan_clear_iro_siege_flags = yes
		set_variable = { var = ironforge_siege_days value = 0 }
	}

	option = {
		name = khaz_modan_siege.10.option.1
		add_war_support = 0.10
		add_stability = 0.05
		custom_effect_tooltip = KMS_iro_siege_begin_tooltip
	}

	option = {
		name = khaz_modan_siege.10.option.2
		add_stability = -0.03
		add_manpower = 2000
		custom_effect_tooltip = KMS_iro_siege_conscript_tooltip
	}
}
