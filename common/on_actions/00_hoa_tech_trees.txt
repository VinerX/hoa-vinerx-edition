on_actions = {
	on_startup = {
		effect = {
			every_possible_country = {
				#Dragon tech
				if = {
					limit = {
						is_dragon_tech = yes
					}
					set_country_flag = dragon_technology
				}
				#Orc tech
				else_if = {
					limit = {
						is_orc_tech = yes
	        		}
					set_country_flag = orc_technology
				}
				#ogre tech
				else_if = {
					limit = {
						is_ogre_tech = yes
					}
					set_country_flag = ogre_technology
				}
				#Troll tech
				else_if = {
					limit = {
						OR = {
							is_troll_tech = yes
							tag = ZLC
						}
	        		}
					set_country_flag = troll_technology
				}
				#Dwarf tech
				else_if = {
					limit = {
	          			is_dwarf_tech = yes
	        		}
					set_country_flag = dwarf_technology
				}
				#High Elf tech
				else_if = {
					limit = {
	          			is_high_elf_tech = yes
	        		}
					set_country_flag = high_elf_technology
				}
				#Night Elf Sentinel tech
				else_if = {
					limit = {
	          			is_night_elf_sentinel_tech = yes
	        		}
					set_country_flag = night_elf_sentinel_technology
				}
				#Night Elf Druid tech
				else_if = {
					limit = {
						is_night_elf_druid_tech = yes
	        		}
					set_country_flag = night_elf_druid_technology
				}
				#Gnoll tech
				else_if = {
					limit = {
	          			is_gnoll_tech = yes
	        		}
					set_country_flag = gnoll_technology
				}
				#quilboar tech
				else_if = {
					limit = {
						is_quilboar_tech = yes
					}
					set_country_flag = quilboar_technology
				}
				#Centaur tech
				else_if = {
					limit = {
	          			is_centaur_tech = yes
	        		}
					set_country_flag = centaur_technology
				}
				#saberon tech
				else_if = {
					limit = {
	          			is_saberon_tech = yes
	        		}
					set_country_flag = saberon_technology
				}
				#Harpy tech
				else_if = {
					limit = {
	          			is_harpy_tech = yes
	        		}
					set_country_flag = harpy_technology
				}
				#Undead tech
				else_if = {
					limit = {
						is_undead_tech = yes
						NOT = { tag = ZLC }
					}
					set_country_flag = undead_technology
				}
				#saurok tech
				else_if = {
					limit = {
	          			is_saurok_tech = yes
	        		}
					set_country_flag = saurok_technology
				}
				#Bog Beast tech
				else_if = {
					limit = {
	          			is_bog_beast_tech = yes
	        		}
					set_country_flag = bog_beast_technology
				}
				#Jade tech
				else_if = {
					limit = {
	          			is_jade_tech = yes
	        		}
					set_country_flag = jade_technology
				}
				#vrykul tech
				else_if = {
					limit = {
	          			is_vrykul_tech = yes
	        		}
					set_country_flag = vrykul_technology
				}
				#nerubian tech
				else_if = {
					limit = {
	          			is_draenei_tech = yes
	        		}
					set_country_flag = draenei_technology
				}
				#bandit tech
				else_if = {
					limit = {
						is_bandits_tech = yes
					}
					set_country_flag = bandits_technology
				}
				#pirates tech
				else_if = {
					limit = {
						is_pirates_tech = yes
					}
					set_country_flag = pirates_technology
				}
				#wildhammer dwarf tech
				else_if = {
					limit = {
						is_wildhammer_dwarf_tech = yes
					}
					set_country_flag = wildhammer_dwarf_technology
				}
				#owb tech
				else_if = {
					limit = {
						is_owb_tech = yes
					}
					set_country_flag = owb_technology
				}
				#Human tech. Fallback/default option
				else = {
					set_country_flag = human_technology
				}
      		}
		}
	}

}
