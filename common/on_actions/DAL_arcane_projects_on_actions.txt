on_actions = {
on_weekly = {
	effect = {
		if = {
			limit = {
				tag = DAL
				has_country_flag = DAL_project_slot_cooldown
				check_variable = { DAL_project_slot_cooldown_days < 14 }
			}
			add_to_variable = { DAL_project_slot_cooldown_days = 1 }
		}
		if = {
			limit = {
				tag = DAL
				has_country_flag = DAL_project_slot_cooldown
				check_variable = { var = DAL_project_slot_cooldown_days value = 14 compare = greater_than_or_equals }
			}
			clr_country_flag = DAL_project_slot_cooldown
			set_variable = { DAL_project_slot_cooldown_days = 0 }
		}
	}
}

on_monthly = {
	effect = {
		if = {
			limit = {
				tag = DAL
				check_variable = { DAL_active_projects < 1 }
				has_country_flag = DAL_strain_stable_ticks
				check_variable = { var = DAL_strain_stable_ticks value = 2 compare = greater_than_or_equals }
				NOT = { dap_strain_stable = yes }
			}
			dap_reduce_strain = yes
			set_variable = { DAL_strain_stable_ticks = 0 }
			clr_country_flag = DAL_strain_stable_ticks
		}
		else_if = {
			limit = {
				tag = DAL
				check_variable = { DAL_active_projects < 1 }
				OR = {
					dap_strain_stretched_trigger = yes
					dap_strain_overload_trigger = yes
					dap_strain_critical_trigger = yes
				}
			}
			if = {
				limit = { has_country_flag = DAL_strain_stable_ticks }
				add_to_variable = { DAL_strain_stable_ticks = 1 }
			}
			else = {
				set_country_flag = DAL_strain_stable_ticks
				set_variable = { DAL_strain_stable_ticks = 1 }
			}
		}
		else = {
			if = {
				limit = { has_country_flag = DAL_strain_stable_ticks }
				clr_country_flag = DAL_strain_stable_ticks
				set_variable = { DAL_strain_stable_ticks = 0 }
			}
		}
	}
}
}
