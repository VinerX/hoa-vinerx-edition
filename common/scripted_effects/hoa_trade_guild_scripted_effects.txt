GEN_upgrade_alchemy = {
	if = {
		limit = { NOT = { has_country_flag = upgrade_alchemy_1 } }
		set_country_flag = upgrade_alchemy_1
	}
	else_if = {
		limit = { has_country_flag = upgrade_alchemy_1 }
