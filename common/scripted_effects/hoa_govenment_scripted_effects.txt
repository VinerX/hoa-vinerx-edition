GEN_upgrade_KINGDOM_exploit = {
	if = {
		limit = { NOT = { has_idea = KINGDOM_exploited_puppet_1 } }
		add_ideas = KINGDOM_exploited_puppet_1
	}
	else_if = {
		limit = { has_idea = KINGDOM_exploited_puppet_1 }
		swap_ideas = {
			remove_idea = KINGDOM_exploited_puppet_1
			add_idea = KINGDOM_exploited_puppet_2
		}
	}
	else_if = {
		limit = { has_idea = KINGDOM_exploited_puppet_2 }
		swap_ideas = {
			remove_idea = KINGDOM_exploited_puppet_2
			add_idea = KINGDOM_exploited_puppet_3
		}
	}
	else_if = {
		limit = { has_idea = KINGDOM_exploited_puppet_3 }
		swap_ideas = {
			remove_idea = KINGDOM_exploited_puppet_3
			add_idea = KINGDOM_exploited_puppet_4
		}
	}
	else_if = {
		limit = { has_idea = KINGDOM_exploited_puppet_4 }
		swap_ideas = {
			remove_idea = KINGDOM_exploited_puppet_4
			add_idea = KINGDOM_exploited_puppet_5
		}
	}
	else_if = {
		limit = { has_idea = KINGDOM_exploited_puppet_5 }
		swap_ideas = {
			remove_idea = KINGDOM_exploited_puppet_5
			add_idea = KINGDOM_exploited_puppet_6
		}
	}
