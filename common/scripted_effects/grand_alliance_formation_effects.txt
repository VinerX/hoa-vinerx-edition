# ============================================================
# Grand Alliance formation — robust faction joining (Second War)
#
# At the 596 (Second War) bookmark the "Alliance of Lordaeron" faction is
# created in history with only STO/STH/HEA/CAE/DAL. The rest of the historical
# roster (KUL, QUE, GIL, STR, GNO, IRO, NRT, NET, SPC, AER) is only written
# into the 600.1.1 history block, which never executes in a 596 campaign.
#
# These effects let the remaining historical members ACTUALLY join the
# Alliance under AI control, instead of only receiving opinion/idea bonuses.
# ============================================================

# ROOT joins the Alliance of Lordaeron faction.
# Robust to faction-leadership changes: prefers LOR, then STO, then any
# surviving core member that currently leads the Alliance faction.
# STO is used as the "anchor" core member because it is always present and
# never leaves the Alliance in the Second War.
join_grand_alliance = {
	if = {
		# Guard: skip if the current country is already allied to the anchor
		# (i.e. already a member of the Alliance).
		limit = { NOT = { is_in_faction_with = STO } }
		# Capture the joining country so it resolves correctly even when this
		# effect is called from within a target/every_country scope.
		save_event_target_as = grand_alliance_joiner
		# Leave any existing faction before joining the Alliance.
		# "leave_faction" covers both regular members and faction leaders,
		# ensuring the Coaliton of Khaz Modan dissolves cleanly.
		if = {
			limit = { is_in_faction = yes }
			leave_faction = yes
		}
		if = {
			limit = { LOR = { exists = yes is_faction_leader = yes } }
			LOR = { add_to_faction = event_target:grand_alliance_joiner }
		}
		else_if = {
			limit = { STO = { exists = yes is_faction_leader = yes } }
			STO = { add_to_faction = event_target:grand_alliance_joiner }
		}
		else = {
			random_country = {
				limit = {
					is_faction_leader = yes
					OR = {
						is_in_faction_with = STO
						tag = STO
					}
				}
				add_to_faction = event_target:grand_alliance_joiner
			}
		}
		alliance_unity_raise_medium = yes
		# If the Second War is already raging (a core Alliance member is at war
		# with the Horde) and the new member is not yet in it, the Horde — the
		# historical aggressor — extends the war to them. This reliably draws the
		# fresh member into the ongoing conflict. If the war has NOT started yet,
		# the member simply joins in peace and is called in defensively later.
		if = {
			limit = {
				NOT = { is_fighting_second_war_horde = yes }
				OR = {
					STO = { is_fighting_second_war_horde = yes }
					LOR = { is_fighting_second_war_horde = yes }
				}
			}
			if = {
				limit = { BRC = { exists = yes is_faction_leader = yes } }
				BRC = { declare_war_on = { target = event_target:grand_alliance_joiner type = annex_everything } }
			}
			else = {
				random_country = {
					limit = {
						is_second_war_horde_clan = yes
						is_faction_leader = yes
					}
					declare_war_on = { target = event_target:grand_alliance_joiner type = annex_everything }
				}
			}
		}
	}
}

# IRO's "Coalition of Khaz Modan" (IRO + GNO + NRT) folds into the Grand
# Alliance. Members are pulled in individually so none are orphaned in a
# leaderless rump faction: partners first, faction leader (IRO) last.
merge_coalition_of_khaz_modan = {
	if = {
		limit = { GNO = { exists = yes } }
		GNO = { join_grand_alliance = yes }
	}
	if = {
		limit = { NRT = { exists = yes } }
		NRT = { join_grand_alliance = yes }
	}
	if = {
		limit = { IRO = { exists = yes } }
		IRO = { join_grand_alliance = yes }
	}
}

# Monthly fallback: if the Coalition merge partially failed
# (e.g. members joined but IRO as faction leader got stuck),
# try again for any remaining Coalition members.
coalition_merge_fallback = {
	if = {
		limit = {
			# The human compact must be functional and the Alliance properly
			# established before Khaz Modan can merge. Otherwise the Coalition
			# stays independent until proper diplomacy (focus branch) resolves.
			LOR = { has_country_flag = human_compact_functional }
			OR = {
				second_war_phase_at_least_2 = yes
				IRO = { has_war_with = BRC }
				GNO = { has_war_with = BRC }
				date > 598.1.1
			}
			OR = {
				AND = {
					IRO = { exists = yes NOT = { is_in_faction_with = STO } is_in_faction = yes }
					any_country = { is_in_faction_with = IRO tag = GNO }
				}
				AND = {
					IRO = { exists = yes NOT = { is_in_faction_with = STO } is_in_faction = yes }
					any_country = { is_in_faction_with = IRO tag = NRT }
				}
			}
		}
		merge_coalition_of_khaz_modan = yes
	}
}

# Pull the smaller alliance-aligned realms (Nethergarde, Stormpike, Aerie
# Peak) into the Grand Alliance if they exist, are alliance-aligned and are
# not committed to a hostile faction. Called from LOR's envoy focus.
consolidate_grand_alliance_minors = {
	every_country = {
		limit = {
			OR = {
				tag = NET
				tag = SPC
				tag = AER
			}
			exists = yes
			has_government = alliance
			NOT = { is_in_faction_with = STO }
			NOT = { has_war_with = STO }
			NOT = { has_war_with = LOR }
		}
		join_grand_alliance = yes
	}
}
