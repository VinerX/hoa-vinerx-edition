nazgrels_fury = {
	if = {
		limit = {
            NOT = { has_idea = OGR_nazgrels_fury_5 }
			NOT = { has_idea = OGR_nazgrels_fury_4 }

			NOT = { has_idea = OGR_nazgrels_fury_2 }
			NOT = { has_idea = OGR_nazgrels_fury_1 }
		}
		custom_effect_tooltip = OGR_nazgrels_fury_tt
	}
	else_if = {
		limit = {
			has_idea = OGR_nazgrels_fury_1
		}
		swap_ideas = {
			remove_idea = OGR_nazgrels_fury_1
			add_idea = OGR_nazgrels_fury_2
		}
	}
	else_if = {
		limit = {
			has_idea = OGR_nazgrels_fury_2
		}

OGR_add_to_burning_insurgency = {
	OGR = {
		add_to_array = {
			array = OGR_burning_insurgency
			value = PREV.id
		}
	}
}

OGR_on_declare_war_burning_insurgency = {
	for_each_scope_loop = {
		array = OGR_burning_insurgency
		if = {
			limit = {
				controller = {
					has_war_with = OGR
				}
				always = yes
			}
			set_state_controller_to = OGR
			remove_dynamic_modifier = {
				modifier = OGR_burning_blade_cult
			}
			OGR = {
				add_to_temp_array = { OGR_burning_insurgency_remove_array = PREV.id }
			}
		}
	}
	for_each_scope_loop = {
		array = OGR_burning_insurgency_remove_array
		OGR = {
			remove_from_array = { OGR_burning_insurgency = PREV.id }
		}
	}
}
