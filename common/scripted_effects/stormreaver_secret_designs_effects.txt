# --- Initialize all Secret Designs variables on SRC ---
src_secret_designs_init = {
	SRC = {
		set_variable = { guldan_suspicion = 15 }
		set_variable = { guldan_forbidden_power = 10 }
		set_variable = { guldan_project_cooldown = 0 }
		set_variable = { guldan_death_knight_count = 0 }
		set_variable = { guldan_tomb_progress = 0 }
		set_country_flag = SRC_secret_designs_unlocked
	}
	src_suspicion_sync = yes
	src_forbidden_power_sync = yes
}

# --- Suspicion manipulation ---
src_suspicion_add_small = {
	set_variable = { guldan_suspicion = guldan_suspicion }
	add_to_variable = { guldan_suspicion = 8 }
	if = { limit = { check_variable = { guldan_suspicion > 100 } } set_variable = { guldan_suspicion = 100 } }
	src_suspicion_sync = yes

# --- Sync suspicion to ideas ---
src_suspicion_sync = {
	SRC = {
		src_suspicion_remove_all = yes
		if = {
			limit = { check_variable = { guldan_suspicion < 20 } }
			add_ideas = SRC_suspicion_trusted
		}
		else_if = {
			limit = { check_variable = { guldan_suspicion < 40 } }
			add_ideas = SRC_suspicion_watched
		}
		else_if = {
			limit = { check_variable = { guldan_suspicion < 60 } }
			add_ideas = SRC_suspicion_suspected
		}
		else_if = {
			limit = { check_variable = { guldan_suspicion < 80 } }
			add_ideas = SRC_suspicion_open_rival
		}
		else = {
			add_ideas = SRC_suspicion_open_rival
		}
	}
}

src_suspicion_remove_all = {
	if = { limit = { has_idea = SRC_suspicion_trusted }		remove_ideas = SRC_suspicion_trusted }
	if = { limit = { has_idea = SRC_suspicion_watched }		remove_ideas = SRC_suspicion_watched }
	if = { limit = { has_idea = SRC_suspicion_suspected }		remove_ideas = SRC_suspicion_suspected }
	if = { limit = { has_idea = SRC_suspicion_open_rival }		remove_ideas = SRC_suspicion_open_rival }
}

# --- Sync Forbidden Power to idea tier ---
src_forbidden_power_sync = {
	SRC = {
		src_forbidden_power_remove_all = yes
		if = {
			limit = { check_variable = { guldan_forbidden_power > 59 } }
			add_ideas = SRC_forbidden_power_high
		}
		else_if = {
			limit = { check_variable = { guldan_forbidden_power > 29 } }
			add_ideas = SRC_forbidden_power_medium
		}
		else = {
			add_ideas = SRC_forbidden_power_low
		}
	}
}

src_forbidden_power_remove_all = {
	if = { limit = { has_idea = SRC_forbidden_power_low }		remove_ideas = SRC_forbidden_power_low }
	if = { limit = { has_idea = SRC_forbidden_power_medium }	remove_ideas = SRC_forbidden_power_medium }
	if = { limit = { has_idea = SRC_forbidden_power_high }		remove_ideas = SRC_forbidden_power_high }
}

# --- Death Knight program progression ---
src_create_death_knights = {
	SRC = {
		if = {
			limit = { NOT = { has_country_flag = SRC_death_knights_created } }
			set_country_flag = SRC_death_knights_created
			add_ideas = SRC_raise_them_all
			if = {
				limit = {
					has_character = SRC_terongor
					NOT = { has_country_flag = SRC_gorefiend_risen }
				}
				set_country_flag = SRC_gorefiend_risen
			}
		}
		else = {
			add_to_variable = { guldan_death_knight_count = 1 }
		}
	}
}

# --- Tomb progress ---
src_tomb_progress_add = {

			set_variable = { guldan_tomb_progress = 6 }
		}
	}
}
