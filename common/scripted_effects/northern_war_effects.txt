# --- Recalculate zone control variables for a country ---
# Use: QUE = { nw_recalc_zone_control = yes } or AMA = { nw_recalc_zone_control = yes }

	# Southern zone: count states controlled
	set_variable = { nw_southern_control = 0 }
	34 = { if = { limit = { is_controlled_by = ROOT } add_to_variable = { nw_southern_control = 1 } } }
	181 = { if = { limit = { is_controlled_by = ROOT } add_to_variable = { nw_southern_control = 1 } } }
	757 = { if = { limit = { is_controlled_by = ROOT } add_to_variable = { nw_southern_control = 1 } } }

	set_variable = { nw_runestone_control = 0 }

	185 = { if = { limit = { is_controlled_by = ROOT } add_to_variable = { nw_runestone_control = 1 } } }
	189 = { if = { limit = { is_controlled_by = ROOT } add_to_variable = { nw_runestone_control = 1 } } }

	107 = { if = { limit = { is_controlled_by = ROOT } add_to_variable = { nw_approaches_control = 1 } } }
	175 = { if = { limit = { is_controlled_by = ROOT } add_to_variable = { nw_approaches_control = 1 } } }

	196 = { if = { limit = { is_controlled_by = ROOT } add_to_variable = { nw_sacred_control = 1 } } }

	107 = { if = { limit = { NOT = { is_controlled_by = ROOT } } add_to_variable = { nw_approaches_enemy = 1 } } }

# --- Mana regeneration tick ---

nw_mana_regen = {
	if = {
		limit = {
			nw_sunwell_controlled = yes
		}
		add_to_variable = { QUE_mana_reserve = 5 }
	}
	# Bonus from Caer Darrow runestone if intact
	if = {
		limit = {
			103 = {
				is_controlled_by = ROOT
				has_state_flag = nw_runestone_caer_darrow_intact
			}
		}
		add_to_variable = { QUE_mana_reserve = 3 }
	}
	# Bonus from intact runestone line
	if = {
		limit = {
			check_variable = { nw_runestones_destroyed = 0 }
		}
		add_to_variable = { QUE_mana_reserve = 2 }
	}
	# Cap at 100
	if = {
		limit = {
			check_variable = { QUE_mana_reserve > 100 }
		}
		set_variable = { QUE_mana_reserve = 100 }
	}
}

# --- Mana drain for barrier maintenance ---

nw_barrier_drain = {
	subtract_from_variable = { QUE_mana_reserve = 10 }
	if = {
		limit = {
			check_variable = { QUE_mana_reserve < 0 }
		}
		set_variable = { QUE_mana_reserve = 0 }
	}
}

# --- Destroy a runestone ---

nw_destroy_runestone = {
	add_to_variable = { nw_runestones_destroyed = 1 }
	subtract_from_variable = { QUE_mana_reserve = 30 }
	if = {
		limit = {
			check_variable = { QUE_mana_reserve < 0 }
		}
		set_variable = { QUE_mana_reserve = 0 }
	}
}

# --- Destroy Caer Darrow runestone ---

nw_destroy_caer_darrow_runestone = {
	103 = {
		set_state_flag = nw_runestone_caer_darrow_destroyed
	}
}
