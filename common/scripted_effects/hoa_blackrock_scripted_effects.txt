# Reduce Horde Overstretch idea (BRC_horde_overstretch_tier_5_idea to BRC_horde_overstretch_tier_1_idea)
reduce_horde_overextension = {
	if = {
		limit = {
			has_idea = BRC_horde_overstretch_tier_5_idea
		}
		swap_ideas = {
			remove_idea = BRC_horde_overstretch_tier_5_idea
			add_idea = BRC_horde_overstretch_tier_4_idea
		}
	}
	else_if = {
		limit = {
			has_idea = BRC_horde_overstretch_tier_4_idea
		}
		swap_ideas = {
			remove_idea = BRC_horde_overstretch_tier_4_idea
			add_idea = BRC_horde_overstretch_tier_3_idea
		}
	}
	else_if = {
		limit = {
			has_idea = BRC_horde_overstretch_tier_3_idea
		}
		swap_ideas = {
			remove_idea = BRC_horde_overstretch_tier_3_idea
			add_idea = BRC_horde_overstretch_tier_2_idea
		}
	}
	else_if = {
		limit = {
			has_idea = BRC_horde_overstretch_tier_2_idea
		}
		swap_ideas = {
			remove_idea = BRC_horde_overstretch_tier_2_idea
			add_idea = BRC_horde_overstretch_tier_1_idea
		}
	}
	else_if = {
		limit = {
			has_idea = BRC_horde_overstretch_tier_1_idea
		}
		remove_ideas = BRC_horde_overstretch_tier_1_idea
	}
}
