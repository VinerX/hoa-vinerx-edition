scripted_gui = {

	tech_count = {
		context_type = player_context
		window_name = "tech_count_container"
		effects = {
			tech_count_bg_click = {
				if = {
					limit = {
						NOT = {
							has_variable = show_tech_count_box
						}
					}
					set_variable = {
						show_tech_count_box = 1
					}
				}
				else = {
					clear_variable = show_tech_count_box
				}
			}
		}
	} 

}
