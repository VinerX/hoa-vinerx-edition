scripted_map_modes = {

	core_map_mode = {
		top = {
			type = state

			color = {
				FROM = {
					set_temp_variable = { alpha = 1.0 }
					if = {
						limit = {
							is_core_of = ROOT
						}
						if = { #core owned by us
							limit = {
								is_controlled_by = ROOT
							}
							set_temp_variable = { red = 0.0 }
							set_temp_variable = { green = 1 }
							set_temp_variable = { blue = 0.0 }
						}
						else_if = { #core occupied by other nation
							limit = {
								is_owned_by = ROOT
							}
							set_temp_variable = { red = 1 }
							set_temp_variable = { green = 0.0 }
							set_temp_variable = { blue = 0.0 }
						}
						else = { #core not owned by us
							set_temp_variable = { red = 0.6 }
							set_temp_variable = { green = 1.0 }
							set_temp_variable = { blue = 0.6 }
						}
					}
					else_if = { #owned non core land
						limit = {
							is_owned_by = ROOT
						}
						if = {
							limit = {
								is_controlled_by = ROOT
							}
							set_temp_variable = { red = 1.0 }
							set_temp_variable = { green = 0.45 }
							set_temp_variable = { blue = 0.0 }
						}
					}
					else_if = { # CLAIMED BY US
						limit = {
							is_claimed_by = ROOT
						}
						set_temp_variable = { red = 1.0 }
						set_temp_variable = { green = 0.0 }
						set_temp_variable = { blue = 1.0 }
					}
					else={
						set_temp_variable = { red = 1}
						set_temp_variable = { green = 0.99 }
						set_temp_variable = { blue = 0.0 }
					}
				}
			}

			thickness = yes

			targets = {
				target_array = global.states
				target_trigger = {
					FROM = {
						OR = {
							is_owned_by = ROOT
							is_controlled_by = ROOT
							is_core_of = ROOT
							is_claimed_by = ROOT
						}
					}
				}
			}
		}

		far_text = country
		near_text = state

		update_daily = no
	}

}
