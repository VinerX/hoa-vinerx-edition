# ----------- LAND TRANSFER DECISIONS -----------
# Temporary per-state decisions visible during BRC_clan_transfer_in_progress.
# Each transfers one Azeroth state to the active arriving clan.
# States: Blasted Lands (26,74) + Redridge Mountains (360,361,362) + Swamp of Sorrows (392)

BRC_transfer_state_26_to_clan = {
	priority = 12
	cost = 0
	fire_only_once = yes

	allowed = { original_tag = BRC }
	visible = {
		has_country_flag = BRC_clan_transfer_in_progress
	}

	available = {
		has_country_flag = BRC_clan_transfer_in_progress
		26 = { is_owned_by = BRC }
	}

	complete_effect = {
		if = {
			limit = { has_country_flag = active_transfer_wsc }
			26 = { transfer_state_to = WSC add_core_of = WSC remove_core_of = BRC }
		}
		else_if = {
			limit = { has_country_flag = active_transfer_smc }
			26 = { transfer_state_to = SMC add_core_of = SMC remove_core_of = BRC }
		}
		else_if = {
			limit = { has_country_flag = active_transfer_tlc }
			26 = { transfer_state_to = TLC add_core_of = TLC remove_core_of = BRC }
		}
		else_if = {
			limit = { has_country_flag = active_transfer_shc }
			26 = { transfer_state_to = SHC add_core_of = SHC remove_core_of = BRC }
		}
		else_if = {
			limit = { has_country_flag = active_transfer_lsc }
			26 = { transfer_state_to = LSC add_core_of = LSC remove_core_of = BRC }
		}
		else_if = {
			limit = { has_country_flag = active_transfer_bcc }
			26 = { transfer_state_to = BCC add_core_of = BCC remove_core_of = BRC }
		}
	}

	ai_will_do = { factor = 0 }
}
