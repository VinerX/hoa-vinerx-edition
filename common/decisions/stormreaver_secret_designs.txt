# --- Raise New Death Knights ---
src_secret_design_death_knights = {
	icon = 1

	available = {
		can_use_secret_designs = yes
		src_guldan_alive_and_leading = yes
		src_in_horde = yes
	}

	visible = {
		can_use_secret_designs = yes
	}

	cost = 25
	days_remove = 90

	complete_effect = {
		SRC = {
			add_timed_idea = { idea = SRC_project_death_knight_research days = 180 }
			if = {
				limit = { NOT = { has_country_flag = SRC_death_knights_created } }
				src_create_death_knights = yes
				add_command_power = 15
				add_manpower = 500
				custom_effect_tooltip = src_create_first_dk_tt
			}
			else = {
				add_to_variable = { guldan_death_knight_count = 1 }
				add_command_power = 10
				custom_effect_tooltip = src_expand_dk_tt
			}
			src_suspicion_reduce_small = yes
		}
	}

	ai_will_do = {
		factor = 1.0
		modifier = {
			factor = 2.0
			has_country_flag = SRC_secret_agenda
		}
		modifier = {
			factor = 1.5
			NOT = { has_country_flag = SRC_death_knights_created }
		}
	}
}
