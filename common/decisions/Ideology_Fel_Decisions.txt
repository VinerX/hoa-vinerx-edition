Fel_agents_of_kiljaden = {

	subjugate_the_weak = {

		available = {
			has_completed_focus = agents_of_kiljaden
			any_neighbor_country = {
				num_of_controlled_states < 4
				NOT = { has_government = fel }
			} 
		}

		visible = {
			has_completed_focus = agents_of_kiljaden
			any_neighbor_country = {
				num_of_controlled_states < 4
				NOT = { has_government = fel }
			} 
		}

		cost = 50

		days_remove = 5

		days_re_enable = 60

		complete_effect = {
			random_neighbor_country = {
				limit = {
					num_of_controlled_states < 4
					NOT = {
						has_government = fel
						is_in_faction_with = ROOT 
						is_subject_of = ROOT
					}
				}
				ROOT = {
					declare_war_on = {
						target = PREV
						type = annex_everything
					}
				}
			}
		}

		ai_will_do = { factor = 1 }
	}
}

						type = annex_everything
