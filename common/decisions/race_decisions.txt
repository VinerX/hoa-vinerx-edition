	accept_ancients_decision = {

		available = {
			any_owned_state = {
				check_variable = { race = 1 }
			}
			NOT = {
				check_variable = { accepted_races^num > 6 }
				has_country_flag = accepting_a_race
			}
		}

		visible = {
			AND = {
				any_owned_state = {
					check_variable = { race = 1 }
				}
				OR = {
					NOT = {
						has_idea = ancients_primary_race
						has_idea = ancients_secondary_race
						check_variable = { accepted_races^num > 6 }
					}
				}
			}
		}

		fire_only_once = no

		cost = 300
		days_remove = 365
		days_re_enable = 365

		remove_effect = {
			add_ideas = {
				ancients_secondary_race
			}
			set_temp_variable = { race_to_accept = 1 }
			add_race_as_accepted_race = yes
			clr_country_flag = accepting_a_race
		}

		complete_effect = {
			set_country_flag = accepting_a_race
		}

		ai_will_do = {
			base = 0
			modifier = {
				add = 1
				has_political_power > 310 #Spend on other things first
			}
		}
	}
