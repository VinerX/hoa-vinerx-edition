DMC_dragon_allocation = {

	DMC_allocate_northern_front = {
		icon = generic_army_bonus
		allowed = { original_tag = DMC }
		available = {
			has_country_flag = DMC_enslaved_dragons
			is_in_faction_with = BRC
			has_political_power > 24
			NOT = { has_country_flag = DMC_northern_front_allocated }
		}
		fire_only_once = yes
		cost = 25
		complete_effect = {
			set_country_flag = DMC_northern_front_allocated

		}
		ai_will_do = { factor = 60 }
	}
