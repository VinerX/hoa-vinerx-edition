	ai_scientist_fix = {
		icon = generic_civil_support
		cost = 0

		ai_will_do = { base = 1 }

		visible = {
			is_ai = yes
			NOT = {	has_country_flag = AI_science }
		}

		available = {
			is_ai = yes
			NOT = {	has_country_flag = AI_science }
		}

		complete_effect = {
			set_country_flag = AI_science

				limit = {
					NOT = { has_scientist_specialization = specialization_land }

					gender = male

				limit = {
