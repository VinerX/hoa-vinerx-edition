trade_guild_decisions = {

	commission_grand_bazaar = {
		icon = GFX_decision_category_generic_industry

		visible = {	
			OR = {
				has_idea = merchant_guild_1
				has_idea = merchant_guild_2
			}
			any_owned_state = {
				state_population_k > 30
				NOT = { has_state_flag = trade_grand_bazaar }
				industrial_complex > 2
			}
		}

		available = { 
			OR = {
				has_idea = merchant_guild_1
				has_idea = merchant_guild_2
			}
			any_owned_state = {
				state_population_k > 30
				NOT = { has_state_flag = trade_grand_bazaar }
				industrial_complex > 7
			}  
		}

		days_remove = 90

		cost = 100

		remove_effect = {	
			random_owned_state = {
				limit = {
					state_population_k > 30
					NOT = { has_state_flag = trade_grand_bazaar }
					industrial_complex > 7
				}
				set_state_flag = trade_grand_bazaar
				add_extra_state_shared_building_slots = 2
				add_building_construction = {
					type = industrial_complex
					level = 1
					instant_build = yes
				}
			}
			if = {
				limit = { NOT = { has_idea = GEN_grand_bazaar_1 } }
				add_ideas = GEN_grand_bazaar_1
			}
			else_if = {
				limit = { has_idea = GEN_grand_bazaar_1 }
				swap_ideas = {
					remove_idea = GEN_grand_bazaar_1
					add_idea = GEN_grand_bazaar_2
				}
			}
			else_if = {
				limit = { has_idea = GEN_grand_bazaar_2 }
				swap_ideas = {
					remove_idea = GEN_grand_bazaar_2
					add_idea = GEN_grand_bazaar_3
				}
			}
		}
		
		ai_will_do = { base = 1 }
	}

	establish_the_crown_market = {
		icon = GFX_decision_category_generic_industry

		visible = {	
			has_idea = merchant_guild_2
		}

		available = { 
			has_idea = merchant_guild_2
			any_owned_state = { has_state_flag = trade_grand_bazaar } 
		}

		days_remove = 30

		cost = 75

		remove_effect = {	
			if = {
				limit = { has_idea = free_trade }
				swap_ideas = {
					remove_idea = free_trade
					add_idea = crown_commerce
				}
			}
			else_if = {
				limit = { has_idea = export_focus }
				swap_ideas = {
					remove_idea = export_focus
					add_idea = crown_commerce
				}
			}
			else_if = {
				limit = { has_idea = limited_exports }
				swap_ideas = {
					remove_idea = limited_exports
					add_idea = crown_commerce
				}
			}
		}
		
		ai_will_do = { base = 1 }
	}
