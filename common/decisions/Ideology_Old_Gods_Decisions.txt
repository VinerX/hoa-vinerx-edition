		available = {
			any_neighbor_country = {
				OR = {

		days_remove = 90

		days_re_enable = 120

		complete_effect = {
			random_neighbor_country = {
				limit = {
					OR = {
						has_idea = red_dragon_primary_race
						has_idea = blue_dragon_primary_race
						has_idea = green_dragon_primary_race
						has_idea = bronze_dragon_primary_race
					}
					NOT = { has_government = old_gods }
				}

				}
				set_country_flag = old_gods_influence
			}
		}

	}
