# Goblins support Horde naval operations — supply ships, low combat ambition
STC_naval_support_horde = {
	allowed = { original_tag = STC }
	enable = {
		is_in_faction_with = BRC
		has_war_with = LOR
	}
	abort_when_not_enabled = yes

	# Low invasion focus — goblins prefer to supply, not to fight
	ai_strategy = {
		type = naval_invasion_focus
		value = 25
	}

	# Steamwheedle is a support fleet, not a second invasion country. In
	# 1.18.3 this priority makes its available ships answer allied invasion
	# support requests, including BRC landings on the mainland.
	ai_strategy = {
		type = naval_invasion_support_priority
		value = 180
	}

	# Keep the far-southern goblin fleet's home bases eligible for the long
	# route to the Horde's central operating waters.
	ai_strategy = {
		type = strike_force_home_base
		id = 1296
		value = 100
	}
	ai_strategy = {
		type = strike_force_home_base
		id = 2134
		value = 80
	}
}
