# Naval invasion priorities against coastal Horde clans (SRC, DMC)
KUL_naval_invasion_horde = {
	allowed = { original_tag = KUL }
	enable = {
		is_in_faction_with = LOR
		has_war_with = BRC
		# Do not split the navy between continental targets before the island
		# chain has created a usable forward network.
		LOR = { check_variable = { alliance_sea_control > 4 } }
	}
	abort_when_not_enabled = yes

	# Fleet bases in main sea zones — not tiny bays
	ai_strategy = {
		type = strike_force_home_base
		id = 58
		value = 100
	}
	ai_strategy = {
		type = strike_force_home_base
		id = 184
		value = 80
	}

	# Invasion focus — Kul Tiras should aggressively prioritize naval invasions
	ai_strategy = {
		type = naval_invasion_focus
		value = 120
	}
	ai_strategy = {
		# 1.18.3 naval AI: make invasion-support assignment outrank
		# routine patrol/escort work while a landing is being prepared.
		type = naval_invasion_support_priority
		value = 160
	}

	# Dragonmaw Port — DMC's main naval base, closest Horde port to KUL
	ai_strategy = {
		type = invasion_unit_request
		state = 503
		value = 100
	}
	ai_strategy = {
		type = invade
		id = "DMC"
		value = 20
	}

	# Stormwind Harbor — SRC's captured naval base, 5 dockyards
	ai_strategy = {
		type = invasion_unit_request
		state = 318
		value = 100
	}
	ai_strategy = {
		type = invade
		id = "SRC"
		value = 20
	}

}
