	allowed = { original_tag = KUL }
	enable = {
		NOT = {	KUL = { iron_mine > 4 } }
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = building_target
		id = iron_mine
		value = 5
	}
}
