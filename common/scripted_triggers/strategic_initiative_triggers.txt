# ============================================================
# Strategic Initiative — scripted triggers
# All read LOR's master variable "strategic_initiative_score".
# ============================================================

# --- Tier checks (exact tier) ---
si_tier_horde_decisive = {
	LOR = { check_variable = { strategic_initiative_score < -5 } }
}
si_tier_horde_initiative = {
	LOR = {
		check_variable = { strategic_initiative_score > -6 }
		check_variable = { strategic_initiative_score < -3 }
	}
}
si_tier_horde_advantage = {
	LOR = {
		check_variable = { strategic_initiative_score > -4 }
		check_variable = { strategic_initiative_score < -1 }
	}
}
si_tier_contested = {

		check_variable = { strategic_initiative_score > -2 }
		check_variable = { strategic_initiative_score < 2 }
	}
}

	}
}

}
