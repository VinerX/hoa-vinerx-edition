# ============================================================
# Draenor Clan Reinforcement Triggers
# Tracks Dark Portal status and clan availability for summoning
# ============================================================

# ----------- Portal Status -----------

# BRC controls province 2854 (the portal) AND state 377, no enemy controls either
portal_route_secure = {
	BRC = {
		controls_state = 377
		NOT = {
			any_enemy_country = {
				controls_state = 377
			}
		}
	}
	NOT = { has_global_flag = dark_portal_sealed }
}

# BRC controls 2854 but enemy controls any state bordering 377
portal_route_threatened = {
	BRC = {
		controls_state = 377
		any_enemy_country = {
			any_controlled_state = {
				is_owned_by = BRC
				OR = {
					state = 376
					state = 378
					state = 379
					state = 380
				}
			}
		}
	}
	NOT = { has_global_flag = dark_portal_sealed }
}

# BRC does NOT control state 377, OR dark_portal_sealed flag is set
portal_route_cut_off = {
	OR = {
		BRC = {
			NOT = { controls_state = 377 }
		}
		has_global_flag = dark_portal_sealed
	}
}

# ----------- General -----------

# BRC can call reinforcements: is faction leader, Second War era, portal not cut off
can_call_draenor_reinforcements = {
	original_tag = BRC
	is_faction_leader = yes
	NOT = { has_start_date > 609.1.1 }
	NOT = { has_start_date < 591.1.1 }
	NOT = { portal_route_cut_off = yes }
}

# Any Draenor clan has outstanding expedition
has_any_pending_expedition = {
	OR = {
		has_country_flag = BRC_WSC_expedition_sent
		has_country_flag = BRC_SMC_expedition_sent
		has_country_flag = BRC_TLC_expedition_sent
		has_country_flag = BRC_SHC_expedition_sent
		has_country_flag = BRC_LSC_expedition_sent
		has_country_flag = BRC_BCC_expedition_sent
	}
}

# Any Draenor clan fully integrated
has_any_clan_integrated = {
	OR = {
		has_country_flag = draenor_clan_summoned_WSC
		has_country_flag = draenor_clan_summoned_SMC
		has_country_flag = draenor_clan_summoned_TLC
		has_country_flag = draenor_clan_summoned_SHC
		has_country_flag = draenor_clan_summoned_LSC
		has_country_flag = draenor_clan_summoned_BCC
	}
}

# Integration cooldown active
is_integration_cooldown_active = {
	OR = {
		has_country_flag = BRC_clan_integration_cooldown
		has_country_flag = BRC_clan_transfer_in_progress
	}
}
