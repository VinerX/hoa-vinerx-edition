# ============================================================
# Second War Phase Controller — scripted triggers
# All read LOR's variable "second_war_phase".
# ============================================================

second_war_phase_is = {
	LOR = { check_variable = { second_war_phase = $phase$ } }
}

second_war_phase_at_least = {
	LOR = { check_variable = { second_war_phase >= $phase$ } }
}

second_war_phase_between = {
	LOR = {
		check_variable = { second_war_phase >= $min$ }
		check_variable = { second_war_phase <= $max$ }
	}
}

second_war_phase_ready_for_1 = {
	OR = {
		has_global_flag = second_war_has_begun
		AND = {
			BRC = { exists = yes has_war_with = LOR }
			LOR = { exists = yes has_war_with = BRC }
		}
		date > 597.1.1
	}
}

second_war_phase_ready_for_2 = {
	OR = {
		AND = {
			LOR = { check_variable = { strategic_initiative_score <= -2 } }
		}
		AND = {
			BRC = {
				any_enemy_country = {
					is_in_faction_with = LOR
					controls_state = 28
				}
			}
		}
		date > 597.7.1
	}
}

second_war_phase_ready_for_3 = {
	OR = {
		AND = {
			AMA = { exists = yes has_war_with = QUE }
		}
		AND = {
			BRC = {
				any_enemy_country = {
					is_in_faction_with = LOR
					controls_state = 34
				}
			}
		}
		date > 598.1.1
	}
}

second_war_phase_ready_for_4 = {
	OR = {
		has_global_flag = the_horde_divided
		AND = {
			LOR = { check_variable = { strategic_initiative_score >= 0 } }
		}
		date > 598.6.1
	}
}

second_war_phase_ready_for_5 = {
	OR = {
		has_global_flag = si_turalyon_rally_milestone
		AND = {
			LOR = { check_variable = { strategic_initiative_score >= 2 } }
		}
		date > 599.1.1
	}
}
