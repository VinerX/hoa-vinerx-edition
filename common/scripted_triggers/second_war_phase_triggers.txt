second_war_phase_is_0 = {
	LOR = { check_variable = { second_war_phase = 0 } }
}

second_war_phase_at_least_1 = {
	LOR = { check_variable = { var = second_war_phase value = 1 compare = greater_than_or_equals } }
}

second_war_phase_at_least_2 = {
	LOR = { check_variable = { var = second_war_phase value = 2 compare = greater_than_or_equals } }
}

second_war_phase_at_least_3 = {
	LOR = { check_variable = { var = second_war_phase value = 3 compare = greater_than_or_equals } }
}

second_war_phase_ready_for_1 = {
	OR = {
		has_global_flag = second_war_has_begun
		AND = {
			BRC = { exists = yes has_war_with = LOR }
			LOR = { exists = yes has_war_with = BRC }
		}
		date > 597.1.1
	}
}

second_war_phase_ready_for_2 = {
	OR = {
		AND = {
			LOR = { check_variable = { var = strategic_initiative_score value = -1 compare = less_than } }
		}
		AND = {
			BRC = {
				any_enemy_country = {
					is_in_faction_with = LOR
					controls_state = 28
				}
			}
		}
		date > 597.7.1
	}
}
