# State trigger to check if the state is owned controlled by ROOT or its puppets
owned_and_controlled_by_ROOT_or_puppets = {
    custom_trigger_tooltip = {
        tooltip = owned_and_controlled_by_ROOT_or_puppets_tt
        OR = {
            is_owned_and_controlled_by = ROOT
            owner = {
                is_subject_of = ROOT
            }
            controller = {
                is_subject_of = ROOT
            }
        }
    }
}

# State trigger to check if the state is owned controlled by ROOT, its puppets or its allies.
owned_and_controlled_by_ROOT_or_puppets_or_allies = {
    custom_trigger_tooltip = {
        tooltip = owned_and_controlled_by_ROOT_or_puppets_or_allies_tt
        OR = {
            is_owned_and_controlled_by = ROOT
            owner = {
              OR = {
                is_subject_of = ROOT
                is_ally_with = ROOT
              }
            }
            controller = {
              OR = {
                is_subject_of = ROOT
                is_ally_with = ROOT
              }
            }
        }
    }
}
