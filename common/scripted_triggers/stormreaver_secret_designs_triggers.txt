# ============================================================

# --- Suspicion checks ---
src_suspicion_trusted = {
	SRC = {
		check_variable = { guldan_suspicion < 20 }
	}
}

src_suspicion_watched = {
	SRC = {
		check_variable = { guldan_suspicion > 19 }
		check_variable = { guldan_suspicion < 40 }
	}
}

src_suspicion_suspected = {
	SRC = {
		check_variable = { guldan_suspicion > 39 }
		check_variable = { guldan_suspicion < 60 }
	}
}

src_suspicion_rival = {
	SRC = {
		check_variable = { guldan_suspicion > 59 }
	}
}

src_suspicion_critical = {
	SRC = {
		check_variable = { guldan_suspicion > 79 }
	}
}

src_suspicion_low = {
	SRC = {
		check_variable = { guldan_suspicion < 40 }
	}
}

src_suspicion_high = {
	SRC = {
		check_variable = { guldan_suspicion > 39 }
	}
}

# --- Forbidden Power checks ---
src_forbidden_power_enough = {
	SRC = {
		check_variable = { guldan_forbidden_power > 14 }
	}
}

src_forbidden_power_small_enough = {
	SRC = {
		check_variable = { guldan_forbidden_power > 7 }
	}
}

src_forbidden_power_high = {
	SRC = {
		check_variable = { guldan_forbidden_power > 59 }
	}
}

# --- Demonic Attention ---
src_demonic_attention_any = {
	SRC = {
		OR = {
			has_country_flag = SRC_demonic_attention_watching
			has_country_flag = SRC_demonic_attention_active
			has_country_flag = SRC_demonic_attention_demanding
		}
	}
}

src_demonic_attention_active = {
	SRC = {
		OR = {
			has_country_flag = SRC_demonic_attention_active
			has_country_flag = SRC_demonic_attention_demanding
		}
	}
}

# --- Warchief Authority integration ---
# High Authority means SRC is more constrained
src_authority_high_constraint = {
	BRC = {
		check_variable = { warchief_authority > 59 }
	}
}

src_authority_low_freedom = {
	BRC = {
		check_variable = { warchief_authority < 40 }
	}
}

# --- Strategic Initiative integration ---
src_si_horde_winning = {
	LOR = { check_variable = { strategic_initiative_score < -3 } }
}

src_si_contested = {
	LOR = {
		check_variable = { strategic_initiative_score > -4 }
		check_variable = { strategic_initiative_score < 4 }
	}
}

src_si_horde_losing = {
	LOR = { check_variable = { strategic_initiative_score > 3 } }
}

# --- Secret Designs category trigger ---
can_use_secret_designs = {
	original_tag = SRC
	SRC = {
		has_country_flag = SRC_secret_designs_unlocked
	}
	NOT = { has_country_flag = SRC_betrayed_horde }
	NOT = { has_country_flag = SRC_guldan_dead }
}

# --- Gul'dan is alive and in charge ---
src_guldan_alive_and_leading = {
	tag = SRC
	has_country_leader = { character = SRC_guldan ruling_only = yes }
	NOT = { has_country_flag = SRC_guldan_dead }
}

# --- Is Gul'dan in the Horde faction ---
src_in_horde = {
	tag = SRC
	is_in_faction_with = BRC
}

# --- Tomb progress triggers ---
src_tomb_progress_started = {
	SRC = {
		check_variable = { guldan_tomb_progress > 0 }
	}
}

src_tomb_progress_mid = {
	SRC = {
		check_variable = { guldan_tomb_progress > 2 }
	}
}

src_tomb_progress_advanced = {
	SRC = {
		check_variable = { guldan_tomb_progress > 4 }
	}
}

# --- Death Knight count ---
src_death_knights_exist = {
	SRC = {
		has_country_flag = SRC_death_knights_created
	}
}

src_death_knights_expanded = {
	SRC = {
		has_country_flag = SRC_death_knights_created
		check_variable = { guldan_death_knight_count > 1 }
	}
}
