# Sea of Blood — island control triggers
# Checks which faction controls a given island state.

# --- State scope: controller is in Alliance of Lordaeron ---
is_island_controlled_by_alliance = {
	controller = {
		is_in_faction_with = LOR
	}
}

# --- State scope: controller is in Orcish Horde ---
is_island_controlled_by_horde = {
	controller = {
		is_in_faction_with = BRC
	}
}

# --- Alliance tier triggers (reads LOR variable) ---
sob_alliance_tier_disputed = {
	LOR = {
		check_variable = { alliance_sea_control < 3 }
	}
}
sob_alliance_tier_weak = {
	LOR = {
		check_variable = { alliance_sea_control = 3 }
	}
}
sob_alliance_tier_contested = {
	LOR = {
		check_variable = { alliance_sea_control = 4 }
	}
}
sob_alliance_tier_advantage = {
	LOR = {
		check_variable = { alliance_sea_control = 5 }
	}
}
sob_alliance_tier_initiative = {
	LOR = {
		check_variable = { alliance_sea_control > 5 }
		check_variable = { alliance_sea_control < 8 }
	}
}
sob_alliance_tier_command = {
	LOR = {
		check_variable = { alliance_sea_control > 7 }
		check_variable = { alliance_sea_control < 10 }
	}
}
sob_alliance_tier_total = {
	LOR = {
		check_variable = { alliance_sea_control = 10 }
	}
}
