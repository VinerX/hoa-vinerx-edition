# ============================================================
# Northern War (Eversong War) — scripted triggers
#
# Zone definitions for the AMA vs QUE woodland campaign.
# Zones are groups of states; triggers check control/ownership.
# ============================================================

# --- Zone ownership checks (fully owned by tag) ---

nw_owns_zone_southern = {
	OR = {
		AND = {
			34 = { is_owned_by = ROOT }
			181 = { is_owned_by = ROOT }
			757 = { is_owned_by = ROOT }
		}
	}
}

		}
	}
}

# --- Zone control checks (controlled by tag via flag) ---

nw_controls_zone_southern = {
	check_variable = { nw_southern_control > 1 }
}

nw_controls_zone_runestone = {
	check_variable = { nw_runestone_control > 1 }
}

nw_controls_zone_farstrider = {
	check_variable = { nw_farstrider_control > 1 }
}

nw_controls_zone_approaches = {
	check_variable = { nw_approaches_control > 1 }
}

nw_controls_zone_sacred = {
	check_variable = { nw_sacred_control > 1 }
}

# --- Zone contested (both sides hold at least one state in zone) ---

nw_zone_southern_contested = {
	check_variable = { nw_southern_control = 1 }
}

	check_variable = { nw_runestone_control = 1 }

}

}

# --- Enemies at the gates (2+ states of approaches zone controlled by enemy) ---

nw_approaches_threatened = {
	check_variable = { nw_approaches_enemy > 1 }
}

# --- Runestone status checks ---

nw_runestone_destroyed_count = {
	has_variable = nw_runestones_destroyed
}

nw_any_runestone_destroyed = {
	check_variable = { nw_runestones_destroyed > 0 }
}

nw_all_runestones_destroyed = {
	check_variable = { nw_runestones_destroyed > 3 }
}

# --- Barrier checks ---

nw_barrier_active = {
	has_country_flag = QUE_ban_dinoriel_active
}

nw_barrier_cooldown = {
	has_country_flag = QUE_ban_dinoriel_cooldown
}

nw_can_activate_barrier = {
	check_variable = { QUE_mana_reserve > 39 }
	NOT = { has_country_flag = QUE_ban_dinoriel_active }
	NOT = { has_country_flag = QUE_ban_dinoriel_cooldown }
}
