# ====================================================================
# Stromgarde Second War — scripted triggers
# ====================================================================

# STR has joined the Alliance via the Second War flow (any commitment level)
STR_has_sw_alliance_commitment = {
	OR = {
		has_country_flag = STR_sw_joined_field_army
		has_country_flag = STR_sw_joined_homeland
		has_country_flag = STR_sw_joined_mountain
	}
}

# STR has a Divided Army spirit active
STR_has_divided_army = {
	OR = {
		has_idea = STR_army_divided_field
		has_idea = STR_army_divided_homeland
		has_idea = STR_army_divided_mountain
	}
}

# Alterac investigation level checks
STR_alterac_investigation_level_0 = {
	NOT = { has_variable = STR_alterac_investigation }
}

STR_alterac_investigation_level_1 = {
	has_variable = STR_alterac_investigation
	check_variable = { STR_alterac_investigation > 0 }
	check_variable = { STR_alterac_investigation < 2 }
}

STR_alterac_investigation_level_2 = {
	check_variable = { STR_alterac_investigation > 1 }
	check_variable = { STR_alterac_investigation < 3 }
}

STR_alterac_investigation_level_3 = {
	check_variable = { STR_alterac_investigation > 2 }
}

STR_alterac_investigation_at_least_2 = {
	check_variable = { STR_alterac_investigation > 1 }
}

# Thandol Span state checks (state 130 is the Thandol Span)
STR_thandol_is_safe = {
	130 = {
		is_controlled_by = STR
	}
}

STR_thandol_is_threatened = {
	any_enemy_country = {
		controls_state = 130
	}
}

# STR is ready for counteroffensive (controls Thandol, war is turning)
STR_ready_for_counteroffensive = {
	is_in_faction_with = LOR
	has_war_with = BRC
	130 = {
		is_controlled_by = STR
	}
	OR = {
		LOR = { has_country_flag = lor_counteroffensive_begun }
		has_global_flag = news_thandol_span_recaptured
	}
}
