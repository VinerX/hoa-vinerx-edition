# ============================================================
# Second War — Horde-state scripted triggers
#
# Replaces brittle single-tag `has_war_with = BRC/BHC` gates in the Alliance
# focus trees. The Second War Horde is the `orcish_horde` faction, anchored on
# BRC. Checking membership of that faction (plus the known clan tags as a
# fallback) keeps focus branches alive even if an individual clan makes a
# separate peace, is destroyed, or the Horde reconfigures.
# ============================================================

# TRUE if the country in scope is a Second War Horde clan.
is_second_war_horde_clan = {
	OR = {
		is_in_faction_with = BRC
		tag = BRC
		tag = BHC
		tag = DMC
		tag = SRC
		tag = TSC
		tag = FWC
		tag = THC
		tag = SMC
		tag = WMC
	}
}

# TRUE if ROOT is currently at war with any member of the Second War Horde.
# Use in `available`/`bypass` instead of `has_war_with = BRC` etc.
is_fighting_second_war_horde = {
	any_country = {
		is_second_war_horde_clan = yes
		has_war_with = ROOT
	}
}
