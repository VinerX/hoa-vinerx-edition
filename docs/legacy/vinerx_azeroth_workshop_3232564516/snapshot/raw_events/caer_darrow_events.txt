country_event = {
	id = cult_of_the_damned.1
	title = cult_of_the_damned.1.t
	desc = cult_of_the_damned.1.d
	# picture = GFX_report_event_german_parade_paris

	is_triggered_only = yes
	fire_only_once = yes

    option = {
    	name = cult_of_the_damned.1.option.1

		hidden_effect = {
			retire_character = CAE_magistrate_marduke

			add_war_support = 0.25
			add_ideas = war_economy

			country_lock_all_division_template = no

			hidden_effect = {
				set_division_template_lock = {
					division_template = "Cult of the Damned Necromancers"
					is_locked = no
				}
				set_division_template_lock = {
					division_template = "Cult of the Damned Skeletons"
					is_locked = no
				}
			}

			load_oob = CAE_andorhal
			load_oob = CAE_scholomance
			if = {
				limit = {
					has_country_flag = andorhall_abomination
				}
				1708 = {
					create_unit = {
						division = "name = \"Cult of the Damned Abominations\" division_template = \"Cult of the Damned Abominations\" start_experience_factor = 0.5 start_equipment_factor = 0.8"
						owner = CAE
						count = 3
					}
				}
				1567 = {
					create_unit = {
						division = "name = \"Cult of the Damned Abominations\" division_template = \"Cult of the Damned Abominations\" start_experience_factor = 0.5 start_equipment_factor = 0.8"
						owner = CAE
						count = 3
					}					
				}
			}
			if = {
				limit = {
					has_country_flag = andorhall_plague_hounds
				}
				1708 = {
					create_unit = {
						division = "name = \"Cult of the Damned Plague Hounds\" division_template = \"Cult of the Damned Plague Hounds\" start_experience_factor = 0.5 start_equipment_factor = 0.8"
						owner = CAE
						count = 3
					}
				}
				1567 = {
					create_unit = {
						division = "name = \"Cult of the Damned Plague Hounds\" division_template = \"Cult of the Damned Plague Hounds\" start_experience_factor = 0.5 start_equipment_factor = 0.8"
						owner = CAE
						count = 3
					}					
				}
			}
		}

		set_politics = {
			ruling_party = death
		}

		add_popularity = {
		    ideology = death
		    popularity = 0.9
		}

		# Accept Undead (id 129)
		set_temp_variable = { race_to_accept = 129 }
		add_race_as_accepted_race = yes
		add_ideas = undead_primary_race

		# Unaccept humans (id 60)
		# (technically there are humans in the cult, but that allows to convert
		# the race of the human conquered lands to undead)
		set_temp_variable = { race_to_accept = 60 }
		remove_race_as_accepted_race = yes

		# Accept Disciples of the Lich King (id 29)
		set_temp_variable = { religion_to_accept = 29 }
		add_religion_as_accepted_religion = yes
		add_ideas = primary_religion_disciples_of_the_lich_king

		# Unaccept Church of the Holy Light (id 1)
		set_temp_variable = { religion_to_unaccept = 1 }
		remove_religion_as_accepted_religion = yes

		every_owned_state = {
			add_core_of = CAE
			set_state_race_to_primary_race = yes
			set_state_religion_to_primary_religion = yes
		}

		swap_ideas = {
			remove_idea = primary_trait_diplomatic
			add_idea = primary_trait_cannibalism
		}
		swap_ideas = {
			remove_idea = primary_trait_human_spirit
			add_idea = primary_trait_no_need_to_sleep
		}
		swap_ideas = {
			remove_idea = primary_trait_perception
			add_idea = primary_trait_dismemberment_irrelevant
		}

		create_wargoal = {
			target = LOR
			type = annex_everything
		}
		declare_war_on = {
			target = LOR
			type = annex_everything
		}
		SRG = { 
			add_to_faction = CAE
			create_wargoal = {
				target = LOR
				type = annex_everything
			}
			declare_war_on = {
				target = LOR
				type = annex_everything
			}
		}
	}
}
